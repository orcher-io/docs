## Forward Recovery Saga

Instead of compensating failed steps (backward recovery), a forward recovery saga retries each step with increasing delays until it eventually succeeds. This approach is appropriate when the operation must eventually complete — for example, when inventory has already been reserved and payment captured, the shipment must be created even if the shipping service is temporarily down. Use durable `ctx.sleep()` timers between retries so the workflow survives worker restarts.

use orcher::prelude::*;
use anyhow::Result;
use std::time::Duration;

// Forward recovery saga - retry until success
#[workflow(name = "forward_recovery_saga", version = "1.0.0")]
pub async fn forward_recovery_saga(
    ctx: WorkflowContext,
    order: OrderData,
) -> Result<OrderResult> {
    println!("Starting forward recovery saga");

    // Step 1: Reserve inventory (retry until success)
    let inventory_id = loop {
        match ctx.execute_task(reserve_inventory, order.items.clone()).await {
            Ok(id) => break id,
            Err(e) => {
                println!("Inventory reservation failed: {}, retrying in 30s", e);
                ctx.sleep(Duration::from_secs(30)).await?;
                continue;
            }
        }
    };

    println!("Inventory reserved: {}", inventory_id);

    // Step 2: Charge payment (retry until success)
    let payment_id = loop {
        match ctx.execute_task(charge_payment, order.amount).await {
            Ok(id) => break id,
            Err(e) => {
                println!("Payment failed: {}, retrying in 60s", e);
                ctx.sleep(Duration::from_secs(60)).await?;
                continue;
            }
        }
    };

    println!("Payment captured: {}", payment_id);

    // Step 3: Create shipment (retry until success)
    let shipment_id = loop {
        match ctx.execute_task(create_shipment, order.order_id.clone()).await {
            Ok(id) => break id,
            Err(e) => {
                println!("Shipment creation failed: {}, retrying in 120s", e);
                ctx.sleep(Duration::from_secs(120)).await?;
                continue;
            }
        }
    };

    println!("Forward recovery saga completed");
    Ok(OrderResult {
        order_id: order.order_id,
        success: true,
        payment_id: Some(payment_id),
        inventory_id: Some(inventory_id),
        shipment_id: Some(shipment_id),
    })
}

from orcher import workflow, WorkflowContext

# Forward recovery saga - retry until success
@workflow(name="forward_recovery_saga", version="1.0.0")
async def forward_recovery_saga(ctx: WorkflowContext, order: dict) -> dict:
    print("Starting forward recovery saga")

    # Step 1: Reserve inventory (retry until success)
    while True:
        try:
            inventory_id = await ctx.execute_task(reserve_inventory, order["items"])
            break
        except Exception as e:
            print(f"Inventory reservation failed: {e}, retrying in 30s")
            await ctx.sleep(30000)

    print(f"Inventory reserved: {inventory_id}")

    # Step 2: Charge payment (retry until success)
    while True:
        try:
            payment_id = await ctx.execute_task(charge_payment, order["amount"])
            break

            await ctx.sleep(60000)

    print(f"Payment captured: {payment_id}")

    # Step 3: Create shipment (retry until success)
    while True:
        try:
            shipment_id = await ctx.execute_task(create_shipment, order["order_id"])
            break
        except Exception as e:

            await ctx.sleep(120000)

    print("Forward recovery saga completed")
    return {

        "success": True,
        "payment_id": payment_id,
        "inventory_id": inventory_id,

    }
