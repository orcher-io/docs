## Parallel Saga with Compensation

When saga steps are independent of each other, you can execute them in parallel to reduce total execution time. If any branch fails, all successful branches are compensated. This pattern is ideal when you need to reserve resources across multiple services simultaneously — such as booking a flight, hotel, and car rental at the same time — where none of the services depend on the others' results.

```rust
use orcher::prelude::*;
use anyhow::Result;
use futures::future::join_all;

#[derive(Debug, Clone, Serialize, Deserialize)]
struct MultiServiceOrder {
    order_id: String,
    services: Vec<String>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
struct ParallelSagaResult {
    order_id: String,
    successful_services: Vec<String>,
    failed_services: Vec<String>,
    success: bool,
}

// Parallel saga with compensation
#[workflow(name = "parallel_saga", version = "1.0.0")]
pub async fn parallel_saga(
    ctx: WorkflowContext,
    order: MultiServiceOrder,
) -> Result<ParallelSagaResult> {
    println!("Starting parallel saga for {}", order.order_id);

    let mut successful = Vec::new();
    let mut failed = Vec::new();
    let mut compensation_ids = Vec::new();

    // Execute all services in parallel
    let mut task_futures = Vec::new();
    for service in &order.services {
        let task = ctx.execute_task(
            execute_service,
            (service.clone(), order.order_id.clone()),
        );
        task_futures.push((service.clone(), task));
    }

    // Wait for all results
    for (service, task) in task_futures {
        match task.await {
            Ok(result_id) => {
                successful.push(service.clone());
                compensation_ids.push((service, result_id));
            }
            Err(e) => {
                println!("Service {} failed: {}", service, e);
                failed.push(service);
            }
        }
    }

    // If any failed, compensate all successful ones
    if !failed.is_empty() {
        println!("Parallel saga failed, compensating {} services", successful.len());

        let mut compensation_tasks = Vec::new();
        for (service, result_id) in compensation_ids {
            let task = ctx.execute_task(
                compensate_service,
                (service, result_id),
            );
            compensation_tasks.push(task);
        }

        let results = join_all(compensation_tasks).await;
        for result in results {
            if let Err(e) = result {
                println!("Compensation failed: {}", e);
            }
        }

        return Ok(ParallelSagaResult {
            order_id: order.order_id,
            successful_services: Vec::new(),
            failed_services: failed,
            success: false,
        });
    }

    println!("Parallel saga completed successfully");
    Ok(ParallelSagaResult {
        order_id: order.order_id,
        successful_services: successful,
        failed_services: failed,
        success: true,
    })
}
```

```typescript
import {
  Workflow, Tasks, Task, WorkflowContext, TaskContext, createTaskRefs
} from '@orcher/sdk';

interface MultiServiceOrder {
  orderId: string;
  services: string[];
}

interface ParallelSagaResult {
  orderId: string;
  successfulServices: string[];
  failedServices: string[];
  success: boolean;
}

@Tasks()
class ParallelSagaTasks {
  @Task({ name: 'executeService' })
  async executeService(
    ctx: TaskContext,
    input: { service: string; orderId: string },
  ): Promise<string> { /* ... */ }

  @Task({ name: 'compensateService' })
  async compensateService(
    ctx: TaskContext,
    input: { service: string; resultId: string },
  ): Promise<void> { /* ... */ }
}

const tasks = createTaskRefs(ParallelSagaTasks);

// Parallel saga with compensation
@Workflow()
class ParallelSaga {
  async execute(
    ctx: WorkflowContext,
    order: MultiServiceOrder,
  ): Promise<ParallelSagaResult> {
    console.log(`Starting parallel saga for ${order.orderId}`);

    const successful: string[] = [];
    const failed: string[] = [];
    const compensationIds: Array<{ service: string; resultId: string }> = [];

    // Execute all services in parallel
    const serviceResults = await Promise.allSettled(
      order.services.map(async (service) => {
        const resultId = await ctx.executeTask(tasks.executeService, {
          service,
          orderId: order.orderId,
        });
        return { service, resultId };
      }),
    );

    for (const result of serviceResults) {
      if (result.status === 'fulfilled') {
        successful.push(result.value.service);
        compensationIds.push(result.value);
      } else {
        failed.push('unknown');
      }
    }

    // If any failed, compensate all successful ones
    if (failed.length > 0) {
      console.log(
        `Parallel saga failed, compensating ${successful.length} services`,
      );

      await Promise.allSettled(
        compensationIds.map(({ service, resultId }) =>
          ctx.executeTask(tasks.compensateService, { service, resultId }),
        ),
      );

      return {
        orderId: order.orderId,
        successfulServices: [],
        failedServices: failed,
        success: false,
      };
    }

    console.log('Parallel saga completed successfully');
    return {
      orderId: order.orderId,
      successfulServices: successful,
      failedServices: failed,
      success: true,
    };
  }
}
```

```python
import asyncio
from orcher import workflow, task, WorkflowContext, TaskContext

@task(name="execute_service")
async def execute_service(ctx: TaskContext, service: str, order_id: str) -> str:
    # ... execute service-specific operation, return result ID
    return f"{service}-result-001"

@task(name="compensate_service")
async def compensate_service(ctx: TaskContext, service: str, result_id: str) -> None:
    # ... compensate a previously executed service operation
    pass

# Parallel saga with compensation
@workflow(name="parallel_saga", version="1.0.0")
async def parallel_saga(ctx: WorkflowContext, order: dict) -> dict:
    print(f"Starting parallel saga for {order['order_id']}")

    successful = []
    failed = []
    compensation_ids = []

    # Execute all services (spawn tasks, then await)
    pending = []
    for service in order["services"]:
        pending.append((service, ctx.execute_task(execute_service, service, order["order_id"])))

    for service, task_future in pending:
        try:
            result_id = await task_future
            successful.append(service)
            compensation_ids.append((service, result_id))
        except Exception as e:
            print(f"Service {service} failed: {e}")
            failed.append(service)

    # If any failed, compensate all successful ones
    if failed:
        print(f"Parallel saga failed, compensating {len(successful)} services")

        for service, result_id in compensation_ids:
            try:
                await ctx.execute_task(compensate_service, service, result_id)
            except Exception as e:
                print(f"Compensation failed: {e}")

        return {
            "order_id": order["order_id"],
            "successful_services": [],
            "failed_services": failed,
            "success": False,
        }

    print("Parallel saga completed successfully")
    return {
        "order_id": order["order_id"],
        "successful_services": successful,
        "failed_services": failed,
        "success": True,
    }
```
