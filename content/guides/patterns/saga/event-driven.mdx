## Event-Driven Saga

In an event-driven saga, each step sends a request to an external service and then waits for a confirmation event before proceeding. This is the right approach when external services are asynchronous — such as a payment gateway that processes charges via webhooks, or a warehouse system that confirms inventory reservations through a message queue. The workflow uses `wait_for_event_with_timeout` to pause until the external system responds, with a timeout to handle cases where the external service never replies.

```rust
use orcher::prelude::*;
use anyhow::Result;
use std::time::Duration;

#[derive(Debug, Clone, Serialize, Deserialize)]
enum SagaEvent {
    InventoryReserved { inventory_id: String },
    PaymentAuthorized { payment_id: String },
    ShipmentCreated { shipment_id: String },
    StepFailed { step: String, reason: String },
}

// Event-driven saga
#[workflow(name = "event_driven_saga", version = "1.0.0")]
pub async fn event_driven_saga(
    ctx: WorkflowContext,
    order: OrderData,
) -> Result<OrderResult> {
    println!("Starting event-driven saga");

    let mut state = SagaState::new(order.order_id.clone());

    // Step 1: Reserve inventory (wait for external confirmation)
    ctx.execute_task(request_inventory_reservation, order.items.clone()).await?;

    let inventory_event: Option<SagaEvent> = ctx
        .wait_for_event_with_timeout("inventory_result", Duration::from_secs(300))
        .await?;

    match inventory_event {
        Some(SagaEvent::InventoryReserved { inventory_id }) => {
            state.inventory_reserved = true;
            state.inventory_id = Some(inventory_id);
        }
        Some(SagaEvent::StepFailed { reason, .. }) => {
            println!("Inventory reservation failed: {}", reason);
            return Ok(OrderResult {
                order_id: order.order_id,
                success: false,
                ..Default::default()
            });
        }
        None => {
            println!("Inventory reservation timed out");
            return Ok(OrderResult {
                order_id: order.order_id,
                success: false,
                ..Default::default()
            });
        }
        _ => unreachable!(),
    }

    // Step 2: Authorize payment (wait for external confirmation)
    ctx.execute_task(request_payment_authorization, order.amount).await?;

    let payment_event: Option<SagaEvent> = ctx
        .wait_for_event_with_timeout("payment_result", Duration::from_secs(300))
        .await?;

    match payment_event {
        Some(SagaEvent::PaymentAuthorized { payment_id }) => {
            state.payment_captured = true;
            state.payment_id = Some(payment_id);
        }
        Some(SagaEvent::StepFailed { reason, .. }) | None => {
            println!("Payment failed, compensating");
            compensate_saga(&ctx, &state).await?;
            return Ok(OrderResult {
                order_id: order.order_id,
                success: false,
                ..Default::default()
            });
        }
        _ => unreachable!(),
    }

    // Step 3: Create shipment (wait for external confirmation)
    ctx.execute_task(request_shipment_creation, order.order_id.clone()).await?;

    let shipment_event: Option<SagaEvent> = ctx
        .wait_for_event_with_timeout("shipment_result", Duration::from_secs(300))
        .await?;

    match shipment_event {
        Some(SagaEvent::ShipmentCreated { shipment_id }) => {
            state.shipment_created = true;
            state.shipment_id = Some(shipment_id);
        }
        Some(SagaEvent::StepFailed { reason, .. }) | None => {
            println!("Shipment failed, compensating");
            compensate_saga(&ctx, &state).await?;
            return Ok(OrderResult {
                order_id: order.order_id,
                success: false,
                ..Default::default()
            });
        }
        _ => unreachable!(),
    }

    println!("Event-driven saga completed");
    Ok(OrderResult {
        order_id: order.order_id,
        success: true,
        payment_id: state.payment_id,
        inventory_id: state.inventory_id,
        shipment_id: state.shipment_id,
    })
}
```

```typescript
import { Workflow, WorkflowContext, createTaskRefs } from '@orcher/sdk';

type SagaEvent =
  | { type: 'InventoryReserved'; inventoryId: string }
  | { type: 'PaymentAuthorized'; paymentId: string }
  | { type: 'ShipmentCreated'; shipmentId: string }
  | { type: 'StepFailed'; step: string; reason: string };

@Tasks()
class EventSagaTasks {
  @Task({ name: 'requestInventoryReservation' })
  async requestInventoryReservation(ctx: TaskContext, items: string[]): Promise<void> { /* ... */ }

  @Task({ name: 'requestPaymentAuthorization' })
  async requestPaymentAuthorization(ctx: TaskContext, amount: number): Promise<void> { /* ... */ }

  @Task({ name: 'requestShipmentCreation' })
  async requestShipmentCreation(ctx: TaskContext, orderId: string): Promise<void> { /* ... */ }
}

const eventTasks = createTaskRefs(EventSagaTasks);

// Event-driven saga
@Workflow()
class EventDrivenSaga {
  async execute(ctx: WorkflowContext, order: OrderData): Promise<OrderResult> {
    console.log('Starting event-driven saga');

    const state = createSagaState(`saga-${order.orderId}`);

    // Step 1: Reserve inventory (wait for external confirmation)
    await ctx.executeTask(eventTasks.requestInventoryReservation, order.items);

    const inventoryEvent = await ctx.waitForEventWithTimeout<SagaEvent>(
      'inventory_result',
      300000,
    );

    if (!inventoryEvent || inventoryEvent.type !== 'InventoryReserved') {
      console.log('Inventory reservation failed or timed out');
      return { orderId: order.orderId, success: false };
    }

    state.inventoryReserved = true;
    state.inventoryId = inventoryEvent.inventoryId;

    // Step 2: Authorize payment (wait for external confirmation)
    await ctx.executeTask(eventTasks.requestPaymentAuthorization, order.amount);

    const paymentEvent = await ctx.waitForEventWithTimeout<SagaEvent>(
      'payment_result',
      300000,
    );

    if (!paymentEvent || paymentEvent.type !== 'PaymentAuthorized') {
      console.log('Payment failed, compensating');
      await compensateSaga(ctx, state);
      return { orderId: order.orderId, success: false };
    }

    state.paymentCaptured = true;
    state.paymentId = paymentEvent.paymentId;

    // Step 3: Create shipment (wait for external confirmation)
    await ctx.executeTask(eventTasks.requestShipmentCreation, order.orderId);

    const shipmentEvent = await ctx.waitForEventWithTimeout<SagaEvent>(
      'shipment_result',
      300000,
    );

    if (!shipmentEvent || shipmentEvent.type !== 'ShipmentCreated') {
      console.log('Shipment failed, compensating');
      await compensateSaga(ctx, state);
      return { orderId: order.orderId, success: false };
    }

    state.shipmentCreated = true;
    state.shipmentId = shipmentEvent.shipmentId;

    console.log('Event-driven saga completed');
    return {
      orderId: order.orderId,
      success: true,
      paymentId: state.paymentId,
      inventoryId: state.inventoryId,
      shipmentId: state.shipmentId,
    };
  }
}
```

```python
@workflow(name="event_driven_saga", version="1.0.0")

    state = {
        "saga_id": f"saga-{order['order_id']}",

        "payment_captured": False, "payment_id": None,
```
