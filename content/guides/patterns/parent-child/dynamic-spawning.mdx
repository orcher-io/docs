## Dynamic Child Spawning

When the number of child workflows isn't known at design time, you can spawn them dynamically based on runtime data. This is the foundation of fan-out patterns — iterate over a collection of items, spawn a child workflow for each one, then collect all results. Use `Promise.allSettled` (TypeScript) or sequential iteration with error handling (Rust/Python) to ensure that one failed child doesn't prevent others from completing.

```rust
use orcher::prelude::*;
use anyhow::Result;

// Child workflow - processes a single invoice
#[workflow(name = "process_invoice", version = "1.0.0")]
pub async fn process_invoice(
    ctx: WorkflowContext,
    invoice_id: String,
) -> Result<InvoiceResult> {
    println!("Processing invoice {}", invoice_id);

    // Generate PDF
    let pdf = ctx.execute_task(generate_pdf, invoice_id.clone()).await?;

    // Send to customer
    ctx.execute_task(send_invoice_email, pdf.clone()).await?;

    Ok(InvoiceResult {
        invoice_id,
        sent: true,
    })
}

// Parent workflow - processes multiple invoices in parallel
#[workflow(name = "monthly_billing", version = "1.0.0")]
pub async fn monthly_billing(
    ctx: WorkflowContext,
    customer_ids: Vec<String>,
) -> Result<BillingResult> {
    println!("Starting monthly billing for {} customers", customer_ids.len());

    // Spawn a child workflow for each customer
    let mut handles = Vec::new();

    for customer_id in customer_ids.iter() {
        let invoice_id = format!("inv-{}-{}", customer_id, ctx.workflow_id());

        let handle = ctx
            .start_child_workflow("process_invoice", invoice_id)
            .await?;

        handles.push(handle);
    }

    println!("Spawned {} invoice workflows", handles.len());

    // Wait for all children to complete
    let mut results = Vec::new();
    for handle in handles {
        match handle.result::<InvoiceResult>().await {
            Ok(result) => results.push(result),
            // Never swallow the suspend signal — re-throw it so the runtime can
            // pause the parent until the child completes.
            Err(e) if e.is_suspend() => return Err(e.into()),
            Err(e) => {
                eprintln!("Invoice failed: {}", e);
                // Continue processing others
            }
        }
    }

    Ok(BillingResult {
        total: results.len(),
        successful: results.iter().filter(|r| r.sent).count(),
        failed: results.len() - results.iter().filter(|r| r.sent).count(),
    })
}
```

```typescript
import {
  Workflow, Tasks, Task, WorkflowContext, TaskContext, createTaskRefs
} from '@orcher/sdk';

interface InvoiceResult {
  invoiceId: string;
  sent: boolean;
}

interface BillingResult {
  total: number;
  successful: number;
  failed: number;
}

@Tasks()
class InvoiceTasks {
  @Task({ name: 'generatePdf' })
  async generatePdf(ctx: TaskContext, invoiceId: string): Promise<PdfData> { /* ... */ }

  @Task({ name: 'sendInvoiceEmail' })
  async sendInvoiceEmail(ctx: TaskContext, pdf: PdfData): Promise<void> { /* ... */ }
}

const tasks = createTaskRefs(InvoiceTasks);

// Child workflow - processes a single invoice
@Workflow()
class ProcessInvoice {
  async execute(ctx: WorkflowContext, invoiceId: string): Promise<InvoiceResult> {
    console.log(`Processing invoice ${invoiceId}`);

    const pdf = await ctx.executeTask(tasks.generatePdf, invoiceId);
    await ctx.executeTask(tasks.sendInvoiceEmail, pdf);

    return { invoiceId, sent: true };
  }
}

// Parent workflow - processes multiple invoices in parallel
@Workflow()
class MonthlyBilling {
  async execute(
    ctx: WorkflowContext,
    customerIds: string[],
  ): Promise<BillingResult> {
    console.log(`Starting monthly billing for ${customerIds.length} customers`);

    // Spawn a child workflow for each customer
    const handles = await Promise.all(
      customerIds.map(async (customerId) => {
        const invoiceId = `inv-${customerId}-${ctx.workflowId()}`;
        return ctx.startChildWorkflow('process_invoice', invoiceId);
      }),
    );

    console.log(`Spawned ${handles.length} invoice workflows`);

    // Wait for all children to complete
    const results = await Promise.allSettled(
      handles.map((handle) => handle.result<InvoiceResult>()),
    );

    const successful = results.filter(
      (r) => r.status === 'fulfilled' && r.value.sent,
    ).length;

    return {
      total: results.length,
      successful,
      failed: results.length - successful,
    };
  }
}
```

```python
from orcher import workflow, task, WorkflowContext, TaskContext

@task(name="generate_pdf")
async def generate_pdf(ctx: TaskContext, invoice_id: str) -> dict:
    # ... generate invoice PDF
    return {"pdf_url": f"https://storage.example.com/{invoice_id}.pdf"}

@task(name="send_invoice_email")
async def send_invoice_email(ctx: TaskContext, pdf: dict) -> None:
    # ... email the invoice to the customer
    pass

# Child workflow - processes a single invoice
@workflow(name="process_invoice", version="1.0.0")
async def process_invoice(ctx: WorkflowContext, invoice_id: str) -> dict:
    print(f"Processing invoice {invoice_id}")

    pdf = await ctx.execute_task(generate_pdf, invoice_id=invoice_id)
    await ctx.execute_task(send_invoice_email, pdf=pdf)

    return {"invoice_id": invoice_id, "sent": True}

# Parent workflow - processes multiple invoices in parallel
@workflow(name="monthly_billing", version="1.0.0")
async def monthly_billing(ctx: WorkflowContext, customer_ids: list[str]) -> dict:
    print(f"Starting monthly billing for {len(customer_ids)} customers")

    # Spawn a child workflow for each customer
    handles = []
    for customer_id in customer_ids:
        invoice_id = f"inv-{customer_id}-{ctx.info.workflow_id}"
        handle = await ctx.start_child_workflow("process_invoice", args=(invoice_id,))
        handles.append(handle)

    print(f"Spawned {len(handles)} invoice workflows")

    # Wait for all children to complete
    successful = 0
    failed = 0
    for handle in handles:
        try:
            result = await handle.result()
            if result["sent"]:
                successful += 1
        except Exception as e:
            print(f"Invoice failed: {e}")
            failed += 1

    return {
        "total": len(handles),
        "successful": successful,
        "failed": failed,
    }
```
