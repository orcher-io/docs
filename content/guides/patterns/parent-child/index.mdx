Parent-child orchestration is a hierarchical pattern where a parent workflow spawns and manages one or more child workflows. Each child runs independently with its own execution context, state, and lifecycle, while the parent coordinates their overall behavior. This pattern lets you build complex systems from simple, reusable building blocks — different teams can own different child workflows, each child has isolated error handling, and you can run hundreds of children in parallel. Use parent-child orchestration when you need to break down complexity, coordinate multiple services, process items in parallel, or enable independent versioning of workflow components.

## Basic Parent-Child Communication

The simplest parent-child pattern spawns a child workflow, passes input data, and waits for the result. The parent calls `start_child_workflow` (or `startChildWorkflow` in TypeScript) with the child's name and input, receiving a handle that can be awaited for the child's return value. This decouples the parent from the child's internal implementation — the parent only needs to know the child's name and its input/output types.

```rust
use orcher::prelude::*;
use anyhow::Result;
use serde::{Deserialize, Serialize};

// Child workflow - validates an order
#[workflow(name = "validate_order", version = "1.0.0")]
pub async fn validate_order(
    ctx: WorkflowContext,
    order: OrderData,
) -> Result<ValidationResult> {
    println!("Validating order {}", order.id);

    // Validate customer
    let customer = ctx.execute_task(check_customer, order.customer_id).await?;

    // Validate inventory
    let inventory = ctx.execute_task(check_inventory, order.items.clone()).await?;

    // Validate pricing
    let pricing = ctx.execute_task(check_pricing, order.total).await?;

    Ok(ValidationResult {
        valid: customer.valid && inventory.available && pricing.correct,
        errors: vec![],
    })
}

// Parent workflow - orchestrates order processing
#[workflow(name = "process_order", version = "1.0.0")]
pub async fn process_order(
    ctx: WorkflowContext,
    order: OrderData,
) -> Result<OrderResult> {
    println!("Processing order {}", order.id);

    // Start validation child workflow
    let validation_handle = ctx
        .start_child_workflow("validate_order", order.clone())
        .await?;

    println!("Validation workflow started: {}", validation_handle.id());

    // Wait for validation result
    let validation = validation_handle.result::<ValidationResult>().await?;

    if !validation.valid {
        return Ok(OrderResult::rejected(order.id, validation.errors));
    }

    // Continue with payment processing
    let payment = ctx.execute_task(process_payment, order.total).await?;

    Ok(OrderResult::success(order.id, payment.transaction_id))
}
```

```typescript
import {
  Workflow, Tasks, Task, WorkflowContext, TaskContext, createTaskRefs
} from '@orcher/sdk';

interface OrderData {
  id: string;
  customerId: string;
  items: string[];
  total: number;
}

interface ValidationResult {
  valid: boolean;
  errors: string[];
}

interface OrderResult {
  orderId: string;
  status: 'success' | 'rejected';
  transactionId?: string;
  errors?: string[];
}

// Tasks used by the child workflow
@Tasks()
class ValidationTasks {
  @Task({ name: 'checkCustomer' })
  async checkCustomer(ctx: TaskContext, customerId: string): Promise<CustomerCheck> { /* ... */ }

  @Task({ name: 'checkInventory' })
  async checkInventory(ctx: TaskContext, items: string[]): Promise<InventoryCheck> { /* ... */ }

  @Task({ name: 'checkPricing' })
  async checkPricing(ctx: TaskContext, total: number): Promise<PricingCheck> { /* ... */ }

  @Task({ name: 'processPayment' })
  async processPayment(ctx: TaskContext, amount: number): Promise<PaymentResult> { /* ... */ }
}

const tasks = createTaskRefs(ValidationTasks);

// Child workflow - validates an order
@Workflow()
class ValidateOrder {
  async execute(ctx: WorkflowContext, order: OrderData): Promise<ValidationResult> {
    console.log(`Validating order ${order.id}`);

    const customer = await ctx.executeTask(tasks.checkCustomer, order.customerId);
    const inventory = await ctx.executeTask(tasks.checkInventory, order.items);
    const pricing = await ctx.executeTask(tasks.checkPricing, order.total);

    return {
      valid: customer.valid && inventory.available && pricing.correct,
      errors: [],
    };
  }
}

// Parent workflow - orchestrates order processing
@Workflow()
class ProcessOrder {
  async execute(ctx: WorkflowContext, order: OrderData): Promise<OrderResult> {
    console.log(`Processing order ${order.id}`);

    // Start validation child workflow
    const validationHandle = await ctx.startChildWorkflow(
      'validate_order',
      order,
    );

    console.log(`Validation workflow started: ${validationHandle.id}`);

    // Wait for validation result
    const validation = await validationHandle.result<ValidationResult>();

    if (!validation.valid) {
      return {
        orderId: order.id,
        status: 'rejected',
        errors: validation.errors,
      };
    }

    // Continue with payment processing
    const payment = await ctx.executeTask(tasks.processPayment, order.total);

    return {
      orderId: order.id,
      status: 'success',
      transactionId: payment.transactionId,
    };
  }
}
```

from orcher import workflow, task, WorkflowContext, TaskContext

# Tasks used by the child workflow
@task(name="check_customer")
async def check_customer(ctx: TaskContext, customer_id: str) -> dict:
    # ... validate customer
    return {"valid": True}

@task(name="check_inventory")
async def check_inventory(ctx: TaskContext, items: list[str]) -> dict:
    # ... check stock levels
    return {"available": True}

@task(name="check_pricing")
async def check_pricing(ctx: TaskContext, total: float) -> dict:
    # ... verify pricing
    return {"correct": True}

@task(name="process_payment")
async def process_payment(ctx: TaskContext, amount: float) -> dict:
    # ... charge payment
    return {"transaction_id": "txn-789"}

# Child workflow - validates an order
@workflow(name="validate_order", version="1.0.0")
async def validate_order(ctx: WorkflowContext, order: dict) -> dict:
    print(f"Validating order {order['id']}")

    customer = await ctx.execute_task(check_customer, order["customer_id"])
    inventory = await ctx.execute_task(check_inventory, order["items"])
    pricing = await ctx.execute_task(check_pricing, order["total"])

    return {
        "valid": customer["valid"] and inventory["available"] and pricing["correct"],
        "errors": [],
    }

# Parent workflow - orchestrates order processing
@workflow(name="process_order", version="1.0.0")
async def process_order(ctx: WorkflowContext, order: dict) -> dict:
    print(f"Processing order {order['id']}")

    # Start validation child workflow
    validation_handle = await ctx.start_child_workflow("validate_order", order)

    print(f"Validation workflow started: {validation_handle.id}")

    # Wait for validation result
    validation = await validation_handle.result()

    if not validation["valid"]:
        return {
            "order_id": order["id"],
            "status": "rejected",
            "errors": validation["errors"],
        }

    # Continue with payment processing
    payment = await ctx.execute_task(process_payment, order["total"])

    return {
        "order_id": order["id"],
        "status": "success",
        "transaction_id": payment["transaction_id"],
    }
