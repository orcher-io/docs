## Parent-Child Event Coordination

Parents can send events (signals) to child workflows to dynamically control their behavior at runtime. This enables patterns like expediting an order, canceling a long-running child, or updating a child's configuration without restarting it. The child calls `wait_for_event` to pause until an event arrives, and the parent (or an external client) sends the event via the child's handle.

```rust
use orcher::prelude::*;
use anyhow::Result;
use std::time::Duration;

// Child workflow that can be expedited
#[workflow(name = "fulfillment_workflow", version = "1.0.0")]
pub async fn fulfillment_workflow(
    ctx: WorkflowContext,
    order_id: String,
) -> Result<FulfillmentResult> {
    println!("Starting standard fulfillment for {}", order_id);

    // Check if expedite event is received
    let expedite: Option<bool> = ctx
        .wait_for_event_with_timeout("expedite", Duration::from_secs(300))
        .await?;

    if expedite.unwrap_or(false) {
        println!("Expedited fulfillment requested!");
        ctx.execute_task(expedited_shipping, order_id).await?;
    } else {
        println!("Standard fulfillment");
        ctx.execute_task(standard_shipping, order_id).await?;
    }

    Ok(FulfillmentResult { completed: true })
}

// Parent workflow that controls child
#[workflow(name = "order_orchestrator", version = "1.0.0")]
pub async fn order_orchestrator(
    ctx: WorkflowContext,
    order: OrderData,
) -> Result<OrderResult> {
    // Start child fulfillment workflow
    let child_handle = ctx
        .start_child_workflow("fulfillment_workflow", order.id.clone())
        .await?;

    println!("Child workflow started: {}", child_handle.id());

    // Wait for external expedite request
    let expedite: Option<bool> = ctx
        .wait_for_event_with_timeout("expedite_request", Duration::from_secs(60))
        .await?;

    if expedite.unwrap_or(false) {
        println!("Expedite requested - notifying child workflow");

        // NOTE: sending events to a child from inside a workflow via the child
        // handle is not yet implemented in the Rust SDK
        // (`ChildWorkflowHandle::send_event` returns an unimplemented error).
        // Deliver the event to the child from an external client instead,
        // addressing it by the child's workflow id.
        child_handle.send_event("expedite", true).await?;
    }

    // Wait for child to complete
    let result = child_handle.result::<FulfillmentResult>().await?;

    Ok(OrderResult {
        order_id: order.id,
        expedited: expedite.unwrap_or(false),
        completed: result.completed,
    })
}
```

```rust
use orcher::client::Client;

async fn expedite_order(workflow_id: &str) -> Result<()> {
    let client = Client::connect("http://localhost:50051").await?;

    // Get a handle to the running parent workflow, then send it an event
    let handle = client.get_workflow_handle(workflow_id).await?;
    handle.send_event("expedite_request", true).await?;

    println!("Expedite request sent");
    Ok(())
}
```

```typescript
import {
  Workflow, Tasks, Task, WorkflowContext, TaskContext, createTaskRefs, Duration
} from '@orcher/sdk';

interface FulfillmentResult {
  completed: boolean;
}

interface OrderResult {
  orderId: string;
  expedited: boolean;
  completed: boolean;
}

@Tasks()
class ShippingTasks {
  @Task({ name: 'expeditedShipping' })
  async expeditedShipping(ctx: TaskContext, orderId: string): Promise<void> { /* ... */ }

  @Task({ name: 'standardShipping' })
  async standardShipping(ctx: TaskContext, orderId: string): Promise<void> { /* ... */ }
}

const tasks = createTaskRefs(ShippingTasks);

// Child workflow that can be expedited
@Workflow()
class FulfillmentWorkflow {
  async execute(ctx: WorkflowContext, orderId: string): Promise<FulfillmentResult> {
    console.log(`Starting standard fulfillment for ${orderId}`);

    // Check if expedite event is received (5 minute timeout).
    // Returns the payload, or null if the timeout elapses first.
    const expedite = await ctx.waitForEventWithTimeout<boolean>(
      'expedite',
      Duration.fromMinutes(5),
    );

    if (expedite) {
      console.log('Expedited fulfillment requested!');
      await ctx.executeTask(tasks.expeditedShipping, orderId);
    } else {
      console.log('Standard fulfillment');
      await ctx.executeTask(tasks.standardShipping, orderId);
    }

    return { completed: true };
  }
}

// Parent workflow that controls child
@Workflow()
class OrderOrchestrator {
  async execute(ctx: WorkflowContext, order: OrderData): Promise<OrderResult> {
    // Start child fulfillment workflow
    const childHandle = await ctx.startChildWorkflow(
      'fulfillment_workflow',
      order.id,
    );

    console.log(`Child workflow started: ${childHandle.id()}`);

    // Wait for external expedite request (1 minute timeout, null on timeout)
    const expedite = await ctx.waitForEventWithTimeout<boolean>(
      'expedite_request',
      Duration.fromMinutes(1),
    );

    if (expedite) {
      console.log('Expedite requested - notifying child workflow');

      // NOTE: ChildWorkflowHandle.sendEvent is not yet implemented in the TS
      // SDK (it throws). Deliver the event to the child from an external client
      // instead, addressing it by the child's workflow id.
      await childHandle.sendEvent('expedite', true);
    }

    // Wait for child to complete
    const result = await childHandle.result<FulfillmentResult>();

    return {
      orderId: order.id,
      expedited: expedite || false,
      completed: result.completed,
    };
  }
}
```

```python
from datetime import timedelta

from orcher import workflow, task, WorkflowContext, TaskContext

@task(name="expedited_shipping")
async def expedited_shipping(ctx: TaskContext, order_id: str) -> None:
    # ... handle expedited shipping
    pass

@task(name="standard_shipping")
async def standard_shipping(ctx: TaskContext, order_id: str) -> None:
    # ... handle standard shipping
    pass

# Child workflow that can be expedited
@workflow(name="fulfillment_workflow", version="1.0.0")
async def fulfillment_workflow(ctx: WorkflowContext, order_id: str) -> dict:
    print(f"Starting standard fulfillment for {order_id}")

    # Check if expedite event is received (5 minute timeout).
    # Python raises TimeoutError when the timeout elapses before the event.
    try:
        expedite = await ctx.wait_for_event_with_timeout(
            "expedite", timedelta(minutes=5)
        )
    except TimeoutError:
        expedite = None

    if expedite:
        print("Expedited fulfillment requested!")
        await ctx.execute_task(expedited_shipping, order_id=order_id)
    else:
        print("Standard fulfillment")
        await ctx.execute_task(standard_shipping, order_id=order_id)

    return {"completed": True}

# Parent workflow that controls child
@workflow(name="order_orchestrator", version="1.0.0")
async def order_orchestrator(ctx: WorkflowContext, order: dict) -> dict:
    # Start child fulfillment workflow
    child_handle = await ctx.start_child_workflow(
        "fulfillment_workflow", args=(order["id"],)
    )

    print(f"Child workflow started: {child_handle.workflow_id}")

    # Wait for external expedite request (1 minute timeout)
    try:
        expedite = await ctx.wait_for_event_with_timeout(
            "expedite_request", timedelta(minutes=1)
        )
    except TimeoutError:
        expedite = None

    if expedite:
        print("Expedite requested - notifying child workflow")
        await child_handle.send_event("expedite", True)

    # Wait for child to complete
    result = await child_handle.result()

    return {
        "order_id": order["id"],
        "expedited": bool(expedite),
        "completed": result["completed"],
    }
```
