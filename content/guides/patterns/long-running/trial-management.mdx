## Trial Period Management

SaaS trial periods are a natural fit for long-running workflows. The workflow sleeps through the trial duration, waking at key milestones to send reminder emails, and at the end checks whether the user has converted to a paid plan or needs to be downgraded.

use orcher::prelude::*;
use anyhow::Result;
use std::time::Duration;

#[derive(Debug, Clone, Serialize, Deserialize)]
struct TrialUser {
    user_id: String,
    email: String,
    trial_days: u64,
    plan: String,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
struct TrialResult {
    user_id: String,
    converted: bool,
    cancelled: bool,
    reminders_sent: u32,
}

#[task]
async fn send_trial_reminder(user_id: String, stage: String) -> Result<()> {
    // Send trial reminder email
    Ok(())
}

#[task]
async fn check_user_status(user_id: String) -> Result<bool> {
    // Check if user upgraded during trial
    Ok(false)
}

#[task]
async fn downgrade_to_free(user_id: String) -> Result<()> {
    // Downgrade user to free plan
    Ok(())
}

// Trial period workflow with reminders
#[workflow(name = "trial_period", version = "1.0.0")]
pub async fn trial_period(
    ctx: WorkflowContext,
    user: TrialUser,
) -> Result<TrialResult> {
    println!("Trial started for user {}: {} days", user.user_id, user.trial_days);

    let mut reminders_sent = 0;
    let trial_duration = Duration::from_secs(user.trial_days * 86400);

    // Send reminder at 50% of trial
    ctx.sleep(trial_duration / 2).await?;
    ctx.execute_task(send_trial_reminder, (user.user_id.clone(), "halfway".to_string())).await?;
    reminders_sent += 1;

    // Send reminder at 80% of trial
    ctx.sleep(trial_duration * 3 / 10).await?;
    ctx.execute_task(send_trial_reminder, (user.user_id.clone(), "ending_soon".to_string())).await?;
    reminders_sent += 1;

    // Send final reminder 1 day before end
    ctx.sleep(trial_duration / 10 - Duration::from_secs(86400)).await?;
    ctx.execute_task(send_trial_reminder, (user.user_id.clone(), "last_day".to_string())).await?;
    reminders_sent += 1;

    // Wait for remaining trial period
    ctx.sleep(Duration::from_secs(86400)).await?;

    // Check if user upgraded during trial
    let upgraded: bool = ctx
        .execute_task(check_user_status, user.user_id.clone())
        .await?;

    if upgraded {
        println!("User {} converted to paid plan", user.user_id);
        Ok(TrialResult {
            user_id: user.user_id,
            converted: true,
            cancelled: false,
            reminders_sent,
        })
    } else {
        println!("Trial expired for user {} - downgrading to free plan", user.user_id);
        ctx.execute_task(downgrade_to_free, user.user_id.clone()).await?;

        Ok(TrialResult {
            user_id: user.user_id,
            converted: false,
            cancelled: false,
            reminders_sent,
        })
    }
}

import { Workflow, Tasks, Task, WorkflowContext, createTaskRefs, Duration } from '@orcher/sdk';

interface TrialUser {
  userId: string;
  email: string;
  trialDays: number;
  plan: string;
}

interface TrialResult {
  userId: string;
  converted: boolean;
  cancelled: boolean;
  remindersSent: number;
}

@Tasks()
class TrialTasks {
  @Task()
  async sendTrialReminder(userId: string, stage: string): Promise<void> {
    // Send trial reminder email
  }

  @Task()
  async checkUserStatus(userId: string): Promise<boolean> {
    // Check if user upgraded during trial
    return false;
  }

  @Task()
  async downgradeToFree(userId: string): Promise<void> {
    // Downgrade user to free plan
  }
}

const tasks = createTaskRefs(TrialTasks);

// Trial period workflow with reminders
@Workflow({ name: 'trial_period', version: '1.0.0' })
class TrialPeriodWorkflow {
  async run(ctx: WorkflowContext, user: TrialUser): Promise<TrialResult> {
    console.log(`Trial started for user ${user.userId}: ${user.trialDays} days`);

    let remindersSent = 0;
    const trialDuration = user.trialDays * 86400000; // milliseconds

    // Send reminder at 50% of trial
    await ctx.sleep(trialDuration / 2);
    await ctx.executeTask(tasks.sendTrialReminder, user.userId, 'halfway');
    remindersSent++;

    // Send reminder at 80% of trial
    await ctx.sleep(trialDuration * 0.3);
    await ctx.executeTask(tasks.sendTrialReminder, user.userId, 'ending_soon');
    remindersSent++;

    // Send final reminder 1 day before end
    await ctx.sleep(trialDuration * 0.1 - 86400000);
    await ctx.executeTask(tasks.sendTrialReminder, user.userId, 'last_day');
    remindersSent++;

    // Wait for remaining trial period
    await ctx.sleep(Duration.fromDays(1));

    // Check if user upgraded during trial
    const upgraded = await ctx.executeTask<boolean>(tasks.checkUserStatus, user.userId);

    if (upgraded) {
      console.log(`User ${user.userId} converted to paid plan`);
      return {
        userId: user.userId,
        converted: true,
        cancelled: false,
        remindersSent,
      };
    }

    console.log(`Trial expired for user ${user.userId} - downgrading to free plan`);
    await ctx.executeTask(tasks.downgradeToFree, user.userId);

    return {
      userId: user.userId,
      converted: false,
      cancelled: false,
      remindersSent,
    };
  }
}

    pass

async def check_user_status(user_id: str) -> bool:

    pass
