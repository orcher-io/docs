Long-running processes are workflows that execute over extended time periods — from hours to days to weeks — while maintaining full reliability and consistency. Orcher's durable timers let you sleep for days without consuming resources, pause for human approvals with automatic timeout and escalation, schedule future actions at precise intervals, and maintain workflow state across restarts and failures. These patterns are ideal whenever you need to wait for external events (human approvals, webhook callbacks, partner responses), schedule future actions (send a reminder in 24 hours, expire a trial in 7 days), implement timeouts (wait for payment confirmation with a 30-minute deadline), coordinate multi-day processes (order fulfillment, employee onboarding, loan approval), or handle recurring tasks (daily reports, weekly cleanups, monthly billing). Because sleeping workflows consume almost no resources, you can run thousands of concurrent long-running workflows with minimal overhead.

## Durable Timers

Durable timers let your workflow sleep for extended periods — hours, days, or even weeks — without consuming any resources. The timer is persisted to the journal, so it survives server restarts, deployments, and failures. When the timer fires, the workflow resumes exactly where it left off.

```rust
use orcher::prelude::*;
use anyhow::Result;
use std::time::Duration;
use serde::{Deserialize, Serialize};

#[derive(Debug, Clone, Serialize, Deserialize)]
struct ReminderInput {
    user_id: String,
    message: String,
    delay_hours: u64,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
struct ReminderResult {
    user_id: String,
    sent: bool,
    sent_at: String,
}

#[task]
async fn send_email_reminder(input: ReminderInput) -> Result<()> {
    // Send the email reminder
    Ok(())
}

#[task]
async fn send_reminder(message: String) -> Result<()> {
    // Send a reminder message
    Ok(())
}

// Send a reminder after a delay
#[workflow(name = "delayed_reminder", version = "1.0.0")]
pub async fn delayed_reminder(
    ctx: WorkflowContext,
    input: ReminderInput,
) -> Result<ReminderResult> {
    println!("Reminder scheduled for {} in {} hours",
        input.user_id, input.delay_hours);

    // Sleep for the specified duration (durable - survives restarts)
    ctx.sleep(Duration::from_secs(input.delay_hours * 3600)).await?;

    println!("Timer fired - sending reminder to {}", input.user_id);

    // Send the reminder
    ctx.execute_task(send_email_reminder, input.clone()).await?;

    Ok(ReminderResult {
        user_id: input.user_id,
        sent: true,
        sent_at: chrono::Utc::now().to_rfc3339(),
    })
}

// Multiple reminders at different intervals
#[workflow(name = "multi_reminder", version = "1.0.0")]
pub async fn multi_reminder(
    ctx: WorkflowContext,
    user_id: String,
) -> Result<()> {
    // Send reminder after 1 day
    ctx.sleep(Duration::from_secs(86400)).await?;
    ctx.execute_task(send_reminder, "Day 1 reminder".to_string()).await?;

    // Send reminder after 3 days (additional 2 days)
    ctx.sleep(Duration::from_secs(172800)).await?;
    ctx.execute_task(send_reminder, "Day 3 reminder".to_string()).await?;

    // Send reminder after 7 days (additional 4 days)
    ctx.sleep(Duration::from_secs(345600)).await?;
    ctx.execute_task(send_reminder, "Day 7 reminder".to_string()).await?;

    println!("All reminders sent for user {}", user_id);
    Ok(())
}
```

```typescript
import { Workflow, Tasks, Task, WorkflowContext, createTaskRefs, Duration } from '@orcher/sdk';

interface ReminderInput {
  userId: string;
  message: string;
  delayHours: number;
}

interface ReminderResult {
  userId: string;
  sent: boolean;
  sentAt: string;
}

@Tasks()
class ReminderTasks {
  @Task()
  async sendEmailReminder(input: ReminderInput): Promise<void> {
    // Send the email reminder
  }

  @Task()
  async sendReminder(message: string): Promise<void> {
    // Send a reminder message
  }
}

const tasks = createTaskRefs(ReminderTasks);

// Send a reminder after a delay
@Workflow({ name: 'delayed_reminder', version: '1.0.0' })
class DelayedReminderWorkflow {
  async run(ctx: WorkflowContext, input: ReminderInput): Promise<ReminderResult> {
    console.log(
      `Reminder scheduled for ${input.userId} in ${input.delayHours} hours`
    );

    // Sleep for the specified duration (durable - survives restarts)
    await ctx.sleep(input.delayHours * 3600 * 1000);

    console.log(`Timer fired - sending reminder to ${input.userId}`);

    // Send the reminder
    await ctx.executeTask(tasks.sendEmailReminder, input);

    return {
      userId: input.userId,
      sent: true,
      sentAt: new Date().toISOString(),
    };
  }
}

// Multiple reminders at different intervals
@Workflow({ name: 'multi_reminder', version: '1.0.0' })
class MultiReminderWorkflow {
  async run(ctx: WorkflowContext, userId: string): Promise<void> {
    // Send reminder after 1 day
    await ctx.sleep(Duration.fromDays(1));
    await ctx.executeTask(tasks.sendReminder, 'Day 1 reminder');

    // Send reminder after 3 days (additional 2 days)
    await ctx.sleep(Duration.fromDays(2));
    await ctx.executeTask(tasks.sendReminder, 'Day 3 reminder');

    // Send reminder after 7 days (additional 4 days)
    await ctx.sleep(Duration.fromDays(4));
    await ctx.executeTask(tasks.sendReminder, 'Day 7 reminder');

    console.log(`All reminders sent for user ${userId}`);
  }
}
```

```python
from orcher import workflow, task, WorkflowContext
from dataclasses import dataclass
from datetime import datetime, timedelta

@dataclass
class ReminderInput:
    user_id: str
    message: str
    delay_hours: int

@dataclass
class ReminderResult:
    user_id: str
    sent: bool
    sent_at: str

@task
async def send_email_reminder(input: ReminderInput) -> None:
    """Send the email reminder."""
    pass

@task
async def send_reminder(message: str) -> None:
    """Send a reminder message."""
    pass

# Send a reminder after a delay
@workflow(name="delayed_reminder", version="1.0.0")
async def delayed_reminder(ctx: WorkflowContext, input: ReminderInput) -> ReminderResult:
    print(f"Reminder scheduled for {input.user_id} in {input.delay_hours} hours")

    # Sleep for the specified duration (durable - survives restarts)
    await ctx.sleep(input.delay_hours * 3600)

    print(f"Timer fired - sending reminder to {input.user_id}")

    # Send the reminder
    await ctx.execute_task(send_email_reminder, input)

    return ReminderResult(
        user_id=input.user_id,
        sent=True,
        sent_at=datetime.utcnow().isoformat(),
    )

# Multiple reminders at different intervals
```
