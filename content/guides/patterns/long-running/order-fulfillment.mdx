## Order Fulfillment Process

Order fulfillment is a multi-day process with distinct stages — payment processing, warehouse picking, packing, shipping, and delivery confirmation. Each stage may involve waiting for external systems or physical operations. Orcher handles this naturally by sleeping between stages and waiting for delivery confirmation signals with automatic timeouts.

use orcher::prelude::*;
use anyhow::Result;
use std::time::Duration;

#[derive(Debug, Clone, Serialize, Deserialize)]
struct Order {
    order_id: String,
    customer_id: String,
    items: Vec<String>,
    total: f64,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
struct FulfillmentResult {
    order_id: String,
    status: String,
    days_taken: u64,
    checkpoints: Vec<String>,
}

#[task]
async fn validate_order(order: Order) -> Result<()> { Ok(()) }

#[task]
async fn charge_payment(total: f64) -> Result<()> { Ok(()) }

#[task]
async fn warehouse_pick(items: Vec<String>) -> Result<bool> { Ok(true) }

#[task]
async fn pack_items(order_id: String) -> Result<()> { Ok(()) }

#[task]
async fn create_shipment(order: Order) -> Result<String> { Ok("TRACK-123".to_string()) }

#[task]
async fn send_delivery_notification(customer_id: String) -> Result<()> { Ok(()) }

// Multi-day order fulfillment workflow
#[workflow(name = "order_fulfillment", version = "1.0.0")]
pub async fn order_fulfillment(
    ctx: WorkflowContext,
    order: Order,
) -> Result<FulfillmentResult> {
    let start = chrono::Utc::now();
    let mut checkpoints = Vec::new();

    println!("Order {} - Starting fulfillment", order.order_id);

    // Day 0: Process order
    ctx.execute_task(validate_order, order.clone()).await?;
    ctx.execute_task(charge_payment, order.total).await?;
    checkpoints.push("order_processed".to_string());

    // Day 0: Wait for warehouse to pick items (typically same day)
    ctx.sleep(Duration::from_secs(14400)).await?; // 4 hours

    let pick_result: bool = ctx
        .execute_task(warehouse_pick, order.items.clone())
        .await?;

    if !pick_result {
        // Wait for inventory restock (could take days)
        ctx.sleep(Duration::from_secs(259200)).await?; // 3 days
        ctx.execute_task(warehouse_pick, order.items.clone()).await?;
    }
    checkpoints.push("items_picked".to_string());

    // Day 1: Pack and ship
    ctx.sleep(Duration::from_secs(86400)).await?; // 1 day
    ctx.execute_task(pack_items, order.order_id.clone()).await?;
    let _tracking_id: String = ctx.execute_task(create_shipment, order.clone()).await?;
    checkpoints.push("shipped".to_string());

    // Day 1-5: Transit time
    ctx.sleep(Duration::from_secs(345600)).await?; // 4 days

    // Day 5: Delivery confirmation
    let delivered: Option<bool> = ctx
        .wait_for_event_with_timeout(
            "delivery_confirmed",
            Duration::from_secs(86400) // 1 day grace period
        )
        .await?;

    if delivered.unwrap_or(false) {
        checkpoints.push("delivered".to_string());
        ctx.execute_task(send_delivery_notification, order.customer_id.clone()).await?;
    } else {
        println!("Delivery not confirmed - marking as complete anyway");
        checkpoints.push("delivery_uncertain".to_string());
    }

    let days_taken = (chrono::Utc::now() - start).num_days() as u64;

    Ok(FulfillmentResult {
        order_id: order.order_id,
        status: "completed".to_string(),
        days_taken,
        checkpoints,
    })
}

import { Workflow, Tasks, Task, WorkflowContext, createTaskRefs, Duration } from '@orcher/sdk';

interface Order {
  orderId: string;
  customerId: string;
  items: string[];
  total: number;
}

interface FulfillmentResult {
  orderId: string;
  status: string;
  daysTaken: number;
  checkpoints: string[];
}

@Tasks()
class FulfillmentTasks {
  @Task()
  async validateOrder(order: Order): Promise<void> {}

  @Task()
  async chargePayment(total: number): Promise<void> {}

  @Task()
  async warehousePick(items: string[]): Promise<boolean> { return true; }

  @Task()
  async packItems(orderId: string): Promise<void> {}

  @Task()
  async createShipment(order: Order): Promise<string> { return 'TRACK-123'; }

  @Task()
  async sendDeliveryNotification(customerId: string): Promise<void> {}
}

const tasks = createTaskRefs(FulfillmentTasks);

// Multi-day order fulfillment workflow
@Workflow({ name: 'order_fulfillment', version: '1.0.0' })
class OrderFulfillmentWorkflow {
  async run(ctx: WorkflowContext, order: Order): Promise<FulfillmentResult> {
    const start = Date.now();
    const checkpoints: string[] = [];

    console.log(`Order ${order.orderId} - Starting fulfillment`);

    // Day 0: Process order
    await ctx.executeTask(tasks.validateOrder, order);
    await ctx.executeTask(tasks.chargePayment, order.total);
    checkpoints.push('order_processed');

    // Day 0: Wait for warehouse to pick items (typically same day)
    await ctx.sleep(Duration.fromHours(4)); // 4 hours

    const pickResult = await ctx.executeTask<boolean>(tasks.warehousePick, order.items);

    if (!pickResult) {
      // Wait for inventory restock (could take days)
      await ctx.sleep(Duration.fromDays(3)); // 3 days
      await ctx.executeTask(tasks.warehousePick, order.items);
    }
    checkpoints.push('items_picked');

    // Day 1: Pack and ship
    await ctx.sleep(Duration.fromDays(1)); // 1 day
    await ctx.executeTask(tasks.packItems, order.orderId);
    const trackingId = await ctx.executeTask(tasks.createShipment, order);
    checkpoints.push('shipped');

    // Day 1-5: Transit time
    await ctx.sleep(Duration.fromDays(4)); // 4 days

    // Day 5: Delivery confirmation
    const delivered = await ctx.waitForEventWithTimeout<boolean>(
      'delivery_confirmed',
      86400000 // 1 day grace period
    );

    if (delivered) {
      checkpoints.push('delivered');
      await ctx.executeTask(tasks.sendDeliveryNotification, order.customerId);
    } else {
      console.log('Delivery not confirmed - marking as complete anyway');
      checkpoints.push('delivery_uncertain');
    }

    const daysTaken = Math.floor((Date.now() - start) / 86400000);

    return {
      orderId: order.orderId,
      status: 'completed',
      daysTaken,
      checkpoints,
    };
  }
}

async def order_fulfillment(ctx: WorkflowContext, order: Order) -> FulfillmentResult:

    print(f"Order {order.order_id} - Starting fulfillment")

    # Day 0: Process order
    await ctx.execute_task(validate_order, order)
    await ctx.execute_task(charge_payment, order.total)

    # Day 0: Wait for warehouse to pick items (typically same day)
    await ctx.sleep(timedelta(hours=4))  # 4 hours

    pick_result = await ctx.execute_task(warehouse_pick, order.items)

    if not pick_result:
        # Wait for inventory restock (could take days)

    # Day 1: Pack and ship

    # Day 1-5: Transit time
