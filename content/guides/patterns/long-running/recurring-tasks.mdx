## Recurring Tasks

Recurring task workflows execute operations on a schedule — daily reports, weekly cleanups, hourly health checks — while maintaining state across iterations. Unlike external cron schedulers, the workflow itself owns the schedule, making it easy to add conditional logic, skip iterations, or gracefully shut down.

```rust
use orcher::prelude::*;
use anyhow::Result;
use std::time::Duration;

#[derive(Debug, Clone, Serialize, Deserialize)]
struct RecurringTaskConfig {
    task_name: String,
    interval_hours: u64,
    max_iterations: u32,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
struct RecurringTaskResult {
    task_name: String,
    iterations_completed: u32,
    total_hours: u64,
}

#[task]
async fn execute_recurring_task(task_name: String) -> Result<()> {
    // Execute the recurring task
    Ok(())
}

#[task]
async fn generate_report(report_type: String) -> Result<()> { Ok(()) }

#[task]
async fn email_report(report_type: String) -> Result<()> { Ok(()) }

#[task]
async fn check_report_active(report_type: String) -> Result<bool> { Ok(true) }

// Recurring task workflow
#[workflow(name = "recurring_task", version = "1.0.0")]
pub async fn recurring_task(
    ctx: WorkflowContext,
    config: RecurringTaskConfig,
) -> Result<RecurringTaskResult> {
    println!("Starting recurring task: {} every {} hours",
        config.task_name, config.interval_hours);

    let interval = Duration::from_secs(config.interval_hours * 3600);
    let mut iteration = 0;

    loop {
        // Execute the task
        println!("Executing {} (iteration {})", config.task_name, iteration + 1);
        ctx.execute_task(execute_recurring_task, config.task_name.clone()).await?;

        iteration += 1;

        // Check if we've reached max iterations
        if iteration >= config.max_iterations {
            println!("Reached max iterations ({}) - stopping", config.max_iterations);
            break;
        }

        // Wait until next execution
        println!("Waiting {} hours until next execution", config.interval_hours);
        ctx.sleep(interval).await?;
    }

    Ok(RecurringTaskResult {
        task_name: config.task_name,
        iterations_completed: iteration,
        total_hours: config.interval_hours * iteration as u64,
    })
}

// Daily report generation
#[workflow(name = "daily_report", version = "1.0.0")]
pub async fn daily_report(
    ctx: WorkflowContext,
    report_type: String,
) -> Result<()> {
    loop {
        // Generate report
        println!("Generating daily {} report", report_type);
        ctx.execute_task(generate_report, report_type.clone()).await?;
        ctx.execute_task(email_report, report_type.clone()).await?;

        // Wait 24 hours until next report
        ctx.sleep(Duration::from_secs(86400)).await?;

        // Check if report should continue
        let should_continue: bool = ctx
            .execute_task(check_report_active, report_type.clone())
            .await?;

        if !should_continue {
            println!("Report {} has been disabled - stopping", report_type);
            break;
        }
    }

    Ok(())
}
```

```typescript
import { Workflow, Tasks, Task, WorkflowContext, createTaskRefs, Duration } from '@orcher/sdk';

interface RecurringTaskConfig {
  taskName: string;
  intervalHours: number;
  maxIterations: number;
}

interface RecurringTaskResult {
  taskName: string;
  iterationsCompleted: number;
  totalHours: number;
}

@Tasks()
class RecurringTasks {
  @Task()
  async executeRecurringTask(taskName: string): Promise<void> {
    // Execute the recurring task
  }

  @Task()
  async generateReport(reportType: string): Promise<void> {}

  @Task()
  async emailReport(reportType: string): Promise<void> {}

  @Task()
  async checkReportActive(reportType: string): Promise<boolean> { return true; }
}

const tasks = createTaskRefs(RecurringTasks);

// Recurring task workflow
@Workflow({ name: 'recurring_task', version: '1.0.0' })
class RecurringTaskWorkflow {
  async run(ctx: WorkflowContext, config: RecurringTaskConfig): Promise<RecurringTaskResult> {
    console.log(
      `Starting recurring task: ${config.taskName} every ${config.intervalHours} hours`
    );

    const interval = config.intervalHours * 3600000; // milliseconds
    let iteration = 0;

    while (true) {
      // Execute the task
      console.log(`Executing ${config.taskName} (iteration ${iteration + 1})`);
      await ctx.executeTask(tasks.executeRecurringTask, config.taskName);

      iteration++;

      // Check if we've reached max iterations
      if (iteration >= config.maxIterations) {
        console.log(`Reached max iterations (${config.maxIterations}) - stopping`);
        break;
      }

      // Wait until next execution
      console.log(`Waiting ${config.intervalHours} hours until next execution`);
      await ctx.sleep(interval);
    }

    return {
      taskName: config.taskName,
      iterationsCompleted: iteration,
      totalHours: config.intervalHours * iteration,
    };
  }
}

// Daily report generation
@Workflow({ name: 'daily_report', version: '1.0.0' })
class DailyReportWorkflow {
  async run(ctx: WorkflowContext, reportType: string): Promise<void> {
    while (true) {
      // Generate report
      console.log(`Generating daily ${reportType} report`);
      await ctx.executeTask(tasks.generateReport, reportType);
      await ctx.executeTask(tasks.emailReport, reportType);

      // Wait 24 hours until next report
      await ctx.sleep(Duration.fromDays(1));

      // Check if report should continue
      const shouldContinue = await ctx.executeTask<boolean>(
        tasks.checkReportActive,
        reportType
      );

      if (!shouldContinue) {
        console.log(`Report ${reportType} has been disabled - stopping`);
        break;
      }
    }
  }
}
```
