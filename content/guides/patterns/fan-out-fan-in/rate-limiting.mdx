## Fan-Out with Rate Limiting

When calling external APIs or databases, unbounded parallelism can overwhelm downstream systems — triggering rate limits, connection pool exhaustion, or service degradation. Rate-limited fan-out processes items in fixed-size batches, waiting for each batch to complete before starting the next. This gives you predictable resource usage while still benefiting from parallelism within each batch.

```rust
use orcher::prelude::*;
use anyhow::Result;
use futures::stream::{self, StreamExt};

// Fan-out with concurrency limit
#[workflow(name = "rate_limited_processor", version = "1.0.0")]
pub async fn rate_limited_processor(
    ctx: WorkflowContext,
    input: BatchInput,
) -> Result<BatchResult> {
    const MAX_CONCURRENT: usize = 10;

    println!("Processing {} items with max {} concurrent",
        input.items.len(), MAX_CONCURRENT);

    // Fan-Out with rate limiting using buffered streams
    let results: Vec<ItemResult> = stream::iter(input.items.iter())
        .map(|item| async {
            ctx.execute_task(process_item, item.clone()).await
        })
        .buffer_unordered(MAX_CONCURRENT)
        .collect::<Vec<_>>()
        .await
        .into_iter()
        .collect::<Result<Vec<_>>>()?;

    // Fan-In: Aggregate results
    let total_value = results.iter().map(|r| r.value).sum();

    Ok(BatchResult {
        job_id: input.job_id,
        total_items: results.len(),
        total_value,
        results,
    })
}
```

```typescript
import { Workflow, WorkflowContext, createTaskRefs } from '@orcher/sdk';

// Fan-out with concurrency limit
@Workflow()
class RateLimitedProcessor {
  async run(ctx: WorkflowContext, input: BatchInput): Promise<BatchResult> {
    const MAX_CONCURRENT = 10;

    console.log(
      `Processing ${input.items.length} items with max ${MAX_CONCURRENT} concurrent`,
    );

    // Process in batches to control concurrency
    const results: ItemResult[] = [];

    for (let i = 0; i < input.items.length; i += MAX_CONCURRENT) {
      const batch = input.items.slice(i, i + MAX_CONCURRENT);

      // Fan-Out: Process batch in parallel
      const batchPromises = batch.map((item) =>
        ctx.executeTask(tasks.processItem, item),
      );

      // Fan-In: Wait for batch to complete before starting next
      const batchResults = await Promise.all(batchPromises);
      results.push(...batchResults);
    }

    const totalValue = results.reduce((sum, r) => sum + r.value, 0);

    return {
      jobId: input.jobId,
      totalItems: results.length,
      totalValue,
      results,
    };
  }
}
```

```python
from orcher import workflow, WorkflowContext

# Fan-out with concurrency limit
@workflow(name="rate_limited_processor", version="1.0.0")
async def rate_limited_processor(ctx: WorkflowContext, input: dict) -> dict:
    MAX_CONCURRENT = 10

    print(f"Processing {len(input['items'])} items with max {MAX_CONCURRENT} concurrent")

    results = []

    # Process in batches to control concurrency
    items = input["items"]
    for i in range(0, len(items), MAX_CONCURRENT):
        batch = items[i:i + MAX_CONCURRENT]

        # Fan-Out: Process batch in parallel
        batch_futures = [ctx.execute_task(process_item, item) for item in batch]

        # Fan-In: Wait for batch to complete before starting next
        for future in batch_futures:
            results.append(await future)

    total_value = sum(r["value"] for r in results)

    return {
        "job_id": input["job_id"],
        "total_items": len(results),
        "total_value": total_value,
        "results": results,
    }
```
