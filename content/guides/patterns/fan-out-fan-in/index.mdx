The Fan-Out/Fan-In pattern splits work across multiple parallel operations (fan-out) and then collects and aggregates their results (fan-in). This maximizes throughput by executing independent operations concurrently — the total latency equals the slowest operation rather than the sum of all operations. Use this pattern when you need to process multiple items in a batch, call multiple APIs simultaneously, aggregate data from distributed sources, run independent validation checks in parallel, or perform map-reduce style dataset transformations.

## Basic Fan-Out/Fan-In

The simplest form of fan-out/fan-in dispatches all tasks at once and waits for every result. Each item is processed independently as a separate task, and the workflow collects all results into a single aggregated response. This approach works well when the number of items is moderate (under a few hundred) and you can tolerate all-or-nothing semantics — if any task fails, the workflow fails.

```rust
use orcher::prelude::*;
use anyhow::Result;
use serde::{Deserialize, Serialize};

#[derive(Debug, Clone, Serialize, Deserialize)]
struct BatchInput {
    job_id: String,
    items: Vec<String>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
struct ItemResult {
    item_id: String,
    processed: bool,
    value: i32,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
struct BatchResult {
    job_id: String,
    total_items: usize,
    total_value: i32,
    results: Vec<ItemResult>,
}

// Task to process a single item
#[task(name = "process_item")]
pub async fn process_item(
    _ctx: TaskContext,
    item_id: String,
) -> Result<ItemResult> {
    let value = item_id.len() as i32 * 10;

    Ok(ItemResult {
        item_id,
        processed: true,
        value,
    })
}

// Basic fan-out/fan-in workflow
#[workflow(name = "batch_processor", version = "1.0.0")]
pub async fn batch_processor(
    ctx: WorkflowContext,
    input: BatchInput,
) -> Result<BatchResult> {
    println!("Processing batch {} with {} items", input.job_id, input.items.len());

    // Fan-Out: Start processing all items in parallel
    let mut task_futures = Vec::new();

    for item in &input.items {
        let task = ctx.execute_task(process_item, item.clone());
        task_futures.push(task);
    }

    // Fan-In: Wait for all tasks to complete
    let mut results = Vec::new();
    let mut total_value = 0;

    for task in task_futures {
        let result: ItemResult = task.await?;
        total_value += result.value;
        results.push(result);
    }

    println!("Batch processing complete: {} items, total value: {}",
        results.len(), total_value);

    Ok(BatchResult {
        job_id: input.job_id,
        total_items: results.len(),
        total_value,
        results,
    })
}
```

```typescript
import {
  Workflow, Tasks, Task, WorkflowContext, TaskContext, createTaskRefs
} from '@orcher/sdk';

interface BatchInput {
  jobId: string;
  items: string[];
}

interface ItemResult {
  itemId: string;
  processed: boolean;
  value: number;
}

interface BatchResult {
  jobId: string;
  totalItems: number;
  totalValue: number;
  results: ItemResult[];
}

@Tasks()
class BatchTasks {
  // Task to process a single item
  @Task({ name: 'processItem' })
  async processItem(ctx: TaskContext, itemId: string): Promise<ItemResult> {
    const value = itemId.length * 10;

    return {
      itemId,
      processed: true,
      value,
    };
  }
}

const tasks = createTaskRefs(BatchTasks);

// Basic fan-out/fan-in workflow
@Workflow()
class BatchProcessor {
  async run(ctx: WorkflowContext, input: BatchInput): Promise<BatchResult> {
    console.log(`Processing batch ${input.jobId} with ${input.items.length} items`);

    // Fan-Out: Start processing all items in parallel
    const taskPromises = input.items.map((item) =>
      ctx.executeTask(tasks.processItem, item),
    );

    // Fan-In: Wait for all tasks to complete
    const results = await Promise.all(taskPromises);

    const totalValue = results.reduce((sum, r) => sum + r.value, 0);

    console.log(`Batch processing complete: ${results.length} items, total value: ${totalValue}`);

    return {
      jobId: input.jobId,
      totalItems: results.length,
      totalValue,
      results,
    };
  }
}
```

```python
from orcher import workflow, task, WorkflowContext, TaskContext

# Task to process a single item
@task(name="process_item")
async def process_item(ctx: TaskContext, item_id: str) -> dict:
    value = len(item_id) * 10

    return {
        "item_id": item_id,
        "processed": True,
        "value": value,
    }

# Basic fan-out/fan-in workflow
@workflow(name="batch_processor", version="1.0.0")
async def batch_processor(ctx: WorkflowContext, input: dict) -> dict:
    print(f"Processing batch {input['job_id']} with {len(input['items'])} items")

    # Fan-Out: Start processing all items in parallel
    task_futures = []
    for item in input["items"]:
        task_futures.append(ctx.execute_task(process_item, item))

    # Fan-In: Wait for all tasks to complete
    results = []
    total_value = 0

    for task_future in task_futures:
        result = await task_future
        total_value += result["value"]
        results.append(result)

    print(f"Batch processing complete: {len(results)} items, total value: {total_value}")

    return {
        "job_id": input["job_id"],
        "total_items": len(results),
        "total_value": total_value,
        "results": results,
    }
```
