## Partial Failure Handling

In many batch processing scenarios, you want to continue processing remaining items even when some fail. Rather than aborting the entire fan-out on the first error, you catch individual task failures, record them separately, and report both successful results and failures to the caller. This gives consumers the information they need to retry just the failed items or accept the partial result.

```rust
use orcher::prelude::*;
use anyhow::Result;

#[derive(Debug, Clone, Serialize, Deserialize)]
struct ResilientResult {
    job_id: String,
    successful: Vec<ItemResult>,
    failed: Vec<FailedItem>,
    success_rate: f64,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
struct FailedItem {
    item_id: String,
    error: String,
}

// Fan-out with partial failure handling
#[workflow(name = "resilient_processor", version = "1.0.0")]
pub async fn resilient_processor(
    ctx: WorkflowContext,
    input: BatchInput,
) -> Result<ResilientResult> {
    println!("Processing {} items with failure tolerance", input.items.len());

    let mut successful = Vec::new();
    let mut failed = Vec::new();

    // Fan-Out: Process all items
    let mut task_handles = Vec::new();
    for item in &input.items {
        let task = ctx.execute_task(process_item, item.clone());
        task_handles.push((item.clone(), task));
    }

    // Fan-In: Collect results, handling failures individually
    for (item_id, task) in task_handles {
        match task.await {
            Ok(result) => {
                successful.push(result);
            }
            Err(e) => {
                println!("Item {} failed: {}", item_id, e);
                failed.push(FailedItem {
                    item_id,
                    error: e.to_string(),
                });
            }
        }
    }

    let success_rate = successful.len() as f64 / input.items.len() as f64;

    println!("Processing complete: {} successful, {} failed ({}% success rate)",
        successful.len(), failed.len(), (success_rate * 100.0) as i32);

    Ok(ResilientResult {
        job_id: input.job_id,
        successful,
        failed,
        success_rate,
    })
}
```

```typescript
  jobId: string;
  successful: ItemResult[];

  successRate: number;

@Workflow()
class ResilientProcessor {
  async run(ctx: WorkflowContext, input: BatchInput): Promise<ResilientResult> {
    console.log(`Processing ${input.items.length} items with failure tolerance`);

    // Fan-Out: Process all items, catching individual failures
    const taskResults = await Promise.allSettled(
      input.items.map(async (item) => {
        const result = await ctx.executeTask(tasks.processItem, item);
        return { item, result };
      }),
    );

    // Fan-In: Separate successes from failures
    const successful: ItemResult[] = [];
    const failed: FailedItem[] = [];

    for (let i = 0; i < taskResults.length; i++) {
      const taskResult = taskResults[i];
      if (taskResult.status === 'fulfilled') {
        successful.push(taskResult.value.result);
      } else {
        console.log(`Item ${input.items[i]} failed:`, taskResult.reason);
        failed.push({
          itemId: input.items[i],
          error: String(taskResult.reason),
        });
      }
    }

    const successRate = successful.length / input.items.length;

    console.log(
      `Processing complete: ${successful.length} successful, ${failed.length} failed (${Math.round(successRate * 100)}% success rate)`,
    );

    return {
      jobId: input.jobId,
      successful,
      failed,
      successRate,
    };
  }
}
```

```python
from orcher import workflow, WorkflowContext

# Fan-out with partial failure handling
@workflow(name="resilient_processor", version="1.0.0")
async def resilient_processor(ctx: WorkflowContext, input: dict) -> dict:
    print(f"Processing {len(input['items'])} items with failure tolerance")

    successful = []
    failed = []

    # Fan-Out: Process all items
    task_handles = []
    for item in input["items"]:
        task_handles.append((item, ctx.execute_task(process_item, item)))

    # Fan-In: Collect results, handling failures individually
    for item_id, task_future in task_handles:
        try:
            result = await task_future
            successful.append(result)
        except Exception as e:
            print(f"Item {item_id} failed: {e}")
            failed.append({"item_id": item_id, "error": str(e)})

    success_rate = len(successful) / len(input["items"]) if input["items"] else 0

    print(
        f"Processing complete: {len(successful)} successful, "
        f"{len(failed)} failed ({int(success_rate * 100)}% success rate)"
    )

    return {
        "job_id": input["job_id"],
        "successful": successful,
        "failed": failed,
        "success_rate": success_rate,
    }
```
