## Workflow-to-Workflow Signals

Workflows coordinate by forwarding events to other workflows. In Python, a workflow can send an event to a child via `child_handle.send_event()` (Rust exposes the same on its child handle). In all languages, external systems and API handlers send events through the client-side workflow handle's `send_event()` / `sendEvent()`. This enables coordination patterns where a parent workflow controls child execution, or sibling workflows notify each other of progress.

> **Note:** In the TypeScript SDK, child-handle event forwarding (`childHandle.sendEvent(...)`) is not yet implemented. Forward events to children via the client handle instead, or use Rust/Python for in-workflow child signaling.

```rust
use orcher::prelude::*;
use anyhow::Result;
use std::time::Duration;

#[task]
pub async fn expedited_fulfillment(_ctx: TaskContext, order_id: String) -> Result<()> { /* ... */ }

#[task]
pub async fn standard_fulfillment(_ctx: TaskContext, order_id: String) -> Result<()> { /* ... */ }

// Child workflow that can be expedited via event
#[workflow]
pub async fn fulfillment_workflow(
    ctx: WorkflowContext,
    order_id: String,
) -> Result<FulfillmentResult> {
    // Check for expedite event (5 minute window)
    let expedite: Option<bool> = ctx
        .wait_for_event_with_timeout("expedite", Duration::from_secs(300))
        .await?;

    if expedite.unwrap_or(false) {
        ctx.execute_task(expedited_fulfillment, order_id).await?;
    } else {
        ctx.execute_task(standard_fulfillment, order_id).await?;
    }

    Ok(FulfillmentResult { completed: true })
}

// Parent workflow that controls the child
#[workflow]
pub async fn parent_workflow(
    ctx: WorkflowContext,
    order_id: String,
) -> Result<ParentResult> {
    // Start child workflow
    let child_handle = ctx
        .start_child_workflow("fulfillment_workflow", order_id.clone())
        .await?;

    // Wait for an external expedite request (1 minute window)
    let expedite: Option<bool> = ctx
        .wait_for_event_with_timeout("expedite_request", Duration::from_secs(60))
        .await?;

    if expedite.unwrap_or(false) {
        // Forward the expedite signal to the child workflow
        child_handle.send_event("expedite", true).await?;
    }

    // Wait for child to complete
    let result = child_handle.result::<FulfillmentResult>().await?;

    Ok(ParentResult {
        order_id,
        expedited: expedite.unwrap_or(false),
        child_completed: result.completed,
    })
}
```

```typescript
import { Workflow, Tasks, Task, WorkflowContext, TaskContext, Duration, createTaskRefs } from '@orcher/sdk';

@Tasks({ taskQueue: 'fulfillment' })
class FulfillmentTasks {
  @Task()
  async expeditedFulfillment(ctx: TaskContext, orderId: string): Promise<void> { /* ... */ }

  @Task()
  async standardFulfillment(ctx: TaskContext, orderId: string): Promise<void> { /* ... */ }
}

const tasks = createTaskRefs(FulfillmentTasks);

// Child workflow that can be expedited via event
@Workflow()
class FulfillmentWorkflow {
  async run(ctx: WorkflowContext, orderId: string): Promise<FulfillmentResult> {
    // Check for expedite event (5 minute window)
    const expedite = await ctx.waitForEventWithTimeout<boolean>('expedite', Duration.fromMinutes(5));

    if (expedite) {
      await ctx.executeTask(tasks.expeditedFulfillment, orderId);
    } else {
      await ctx.executeTask(tasks.standardFulfillment, orderId);
    }

    return { completed: true };
  }
}

// Parent workflow that controls the child
@Workflow()
class ParentWorkflow {
  async run(ctx: WorkflowContext, orderId: string): Promise<ParentResult> {
    // Start child workflow
    const childHandle = await ctx.startChildWorkflow('fulfillmentWorkflow', orderId);

    // Wait for an external expedite request (1 minute window)
    const expedite = await ctx.waitForEventWithTimeout<boolean>('expedite_request', Duration.fromMinutes(1));

    if (expedite) {
      // NOTE: child-handle event forwarding is not yet implemented in the
      // TypeScript SDK (childHandle.sendEvent throws "not implemented").
      // Forward events to the child via the client handle instead, or use
      // the Rust/Python SDK for in-workflow child signaling.
    }

    // Wait for child to complete
    const result = await childHandle.result<FulfillmentResult>();

    return {
      orderId,
      expedited: expedite || false,
      childCompleted: result.completed,
    };
  }
}
```

<TabPanel id="python" label="Python">

```python
from datetime import timedelta
from dataclasses import dataclass
from orcher import workflow, task, WorkflowContext, TaskContext


@dataclass
class FulfillmentResult:
    completed: bool


@dataclass
class ParentResult:
    order_id: str
    expedited: bool
    child_completed: bool


@task
async def expedited_fulfillment(ctx: TaskContext, order_id: str) -> None:
    ...

@task
async def standard_fulfillment(ctx: TaskContext, order_id: str) -> None:
    ...


# Child workflow that can be expedited via event
@workflow
async def fulfillment_workflow(ctx: WorkflowContext, order_id: str) -> FulfillmentResult:
    # Check for expedite event (5 minute window)
    expedite = await ctx.wait_for_event_with_timeout("expedite", timedelta(minutes=5))

    if expedite:
        await ctx.execute_task(expedited_fulfillment, arg=order_id)
    else:
        await ctx.execute_task(standard_fulfillment, arg=order_id)

    return FulfillmentResult(completed=True)


# Parent workflow that controls the child
@workflow
async def parent_workflow(ctx: WorkflowContext, order_id: str) -> ParentResult:
    # Start child workflow
    child_handle = await ctx.start_child_workflow("fulfillment_workflow", order_id)

    # Wait for an external expedite request (1 minute window)
    expedite = await ctx.wait_for_event_with_timeout(
        "expedite_request", timedelta(minutes=1)
    )

    if expedite:
        # Forward the expedite signal to the child workflow
        await child_handle.send_event("expedite", True)

    # Wait for child to complete
    result = await child_handle.result()

    return ParentResult(
        order_id=order_id,
        expedited=bool(expedite),
        child_completed=result["completed"],
    )
```

## Event Buffering

Events are automatically buffered if they arrive before the workflow calls `wait_for_event()`. This means the order of "send" and "wait" doesn't matter — the event is never lost. On replay, buffered events are replayed from the journal.
