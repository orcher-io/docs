This page covers two ways to move data: **batch processing** a whole dataset in chunks, and **incremental sync** — only pulling what changed since last time. The first you saw in miniature on the last page; the second is the pattern that turns a nightly full-reload into a lightweight sync that runs every few minutes.

## Batch Processing

You already know this shape from the [CSV import](/guides/learning/data-engineers/etl-pipelines) — count the work, loop over chunks, checkpoint each one. Here it's a **database-to-database migration** instead of a file, with an enrichment step in the middle. Same durability, new source and transform. We'll build it in three moves.

### Step 1 — Page through the source

Two tasks read the source: `count_source_rows` for the total, and `fetch_batch` for one page via `LIMIT/OFFSET` — the cursor pagination you'd write against any large table. Each fetched page is a journaled task, so a crash costs you one page, not the whole migration.

> **Note on `OFFSET`.** `LIMIT/OFFSET` is the clearest way to show pagination, but on very large tables the database re-scans skipped rows, so deep offsets get slow. In production, prefer *keyset* pagination — `WHERE id > $last_id ORDER BY id LIMIT $n` — which is also the seam that leads straight into incremental sync below.

### Step 2 — Enrich, then load idempotently

`transform_rows` does the enrichment the migration exists for — lowercasing emails, mapping a country code to a region, normalizing dates. `upsert_batch` writes with an UPSERT so a retried batch updates in place instead of duplicating — the idempotency rule from the last page, and it never stops mattering.

### Step 3 — Orchestrate the migration

The workflow is the same count-then-loop as before, now fetch → transform → upsert per batch. Nothing tracks progress explicitly — the loop index lives in the journal.

<Tabs scrollable size="small" type="underlined" defaultActiveId="typescript" queryGroup="language">
<TabPanel id="typescript" label="TypeScript">

```typescript
@Workflow({ name: 'database_migration', version: '1.0.0' })
class DatabaseMigration {
  async run(ctx: WorkflowContext, input: MigrationInput): Promise<MigrationResult> {
    const totalRows = await ctx.executeTask(tasks.countSourceRows, input.sourceTable);
    const numBatches = Math.ceil(totalRows / input.batchSize);

    for (let i = 0; i < numBatches; i++) {
      const rows = await ctx.executeTask(tasks.fetchBatch, {
        table: input.sourceTable, offset: i * input.batchSize, limit: input.batchSize,
      });
      const transformed = await ctx.executeTask(tasks.transformRows, rows);
      await ctx.executeTask(tasks.upsertBatch, { table: input.targetTable, rows: transformed });
    }

    return { rowsMigrated: totalRows, batchesProcessed: numBatches };
  }
}
```

#[workflow(name = "database_migration", version = "1.0.0")]
async fn database_migration(ctx: WorkflowContext, input: MigrationInput) -> Result<MigrationResult> {
    let total_rows = ctx.execute_task(count_source_rows, input.source_table.clone()).await?;
    let num_batches = (total_rows as usize + input.batch_size - 1) / input.batch_size;

    for i in 0..num_batches {
        let rows = ctx.execute_task(fetch_batch, (
            input.source_table.clone(), i * input.batch_size, input.batch_size,
        )).await?;
        let transformed = ctx.execute_task(transform_rows, rows).await?;
        ctx.execute_task(upsert_batch, (input.target_table.clone(), transformed)).await?;
    }

    Ok(MigrationResult { rows_migrated: total_rows, batches_processed: num_batches })
}

## Incremental Sync (CDC Pattern)

Re-processing everything on every run doesn't scale — once a table has millions of rows, you want to touch only what changed since last time. That's **change data capture (CDC)**, and you've built it before:

- **A cursor column + cron.** Store `last_synced_at`, `SELECT ... WHERE updated_at > last_synced_at`, advance the cursor. Simple — until a run fails *after* advancing the cursor but *before* loading, and you've silently skipped a window forever.
- **Debezium + Kafka Connect.** Robust, streaming CDC — but a broker, connectors, and a schema registry to operate for what might be a five-minute sync.

The pattern below is the cursor approach with the classic bug engineered out: the cursor (here called a **watermark**) advances **only after** the load succeeds, and because that ordering is journaled, a crash can't leave it half-advanced.

### Step 3 — Load, then advance the watermark

Two tasks, and their **order is the whole pattern**. `load_records` UPSERTs the batch; `save_watermark` records how far we got. Load first: if it fails, the watermark is untouched and the next run re-syncs the same window safely. Save last: the watermark only ever names data that's already landed.

    VALUES ($1, $2, NOW())
    ON CONFLICT (source) DO UPDATE SET watermark = $2, updated_at = NOW()

### Step 4 — Orchestrate the sync

The workflow runs the five tasks in the order that makes the watermark safe, with one shortcut: if nothing's new, return early without touching the watermark.

<Tabs scrollable size="small" type="underlined" defaultActiveId="typescript" queryGroup="language">
<TabPanel id="typescript" label="TypeScript">

```typescript
@Workflow({ name: 'incremental_sync', version: '1.0.0' })
class IncrementalSync {
  async run(ctx: WorkflowContext, input: SyncInput): Promise<SyncResult> {
    const watermark = await ctx.executeTask(tasks.getWatermark, input.sourceName);
    const since = watermark ?? '2024-01-01T00:00:00Z';

    const records = await ctx.executeTask(tasks.fetchNewRecords, { source: input.sourceName, since });
    if (records.length === 0) {
      return { recordsSynced: 0, newWatermark: since }; // nothing new — don't move the watermark
    }

    const transformed = await ctx.executeTask(tasks.transformRecords, records);
    await ctx.executeTask(tasks.loadRecords, { table: input.targetTable, records: transformed });

    // advance the watermark ONLY after the load succeeded
    const newWatermark = records[records.length - 1].created_at;
    await ctx.executeTask(tasks.saveWatermark, { source: input.sourceName, watermark: newWatermark });

    return { recordsSynced: records.length, newWatermark };
  }
}
```

#[workflow(name = "incremental_sync", version = "1.0.0")]
async fn incremental_sync(ctx: WorkflowContext, input: SyncInput) -> Result<SyncResult> {
    let watermark = ctx.execute_task(get_watermark, input.source_name.clone()).await?;
    let since = watermark.unwrap_or_else(|| "2024-01-01T00:00:00Z".into());

    let records = ctx.execute_task(fetch_new_records, (input.source_name.clone(), since.clone())).await?;
    if records.is_empty() {
        return Ok(SyncResult { records_synced: 0, new_watermark: since }); // nothing new
    }

    let transformed = ctx.execute_task(transform_records, records.clone()).await?;
    ctx.execute_task(load_records, (input.target_table, transformed)).await?;

    // advance the watermark ONLY after the load succeeded
    let new_watermark = records.last().unwrap().created_at.clone();
    ctx.execute_task(save_watermark, (input.source_name, new_watermark.clone())).await?;

    Ok(SyncResult { records_synced: records.len(), new_watermark })
}

        amount: r.amount as f64 / 100.0,
        currency: r.currency.to_uppercase(),
        status: map_status(&r.status),
        customer_id: r.customer_id,
        created_at: r.created_at,
    }).collect())
}

#[task]
async fn load_records(ctx: TaskContext, input: (String, Vec<TargetRecord>)) -> Result<usize> {
    let (table, records) = input;
    let pool = get_db_pool().await?;
    db::upsert_rows(&pool, &table, &records).await
}

#[task]
async fn save_watermark(ctx: TaskContext, input: (String, String)) -> Result<()> {
    let (source, watermark) = input;
    let pool = get_db_pool().await?;
    sqlx::query(
        "INSERT INTO sync_state (source, watermark, updated_at)
         VALUES ($1, $2, NOW())
         ON CONFLICT (source) DO UPDATE SET watermark = $2, updated_at = NOW()"
    ).bind(&source).bind(&watermark).execute(&pool).await?;
    Ok(())
}

#[workflow(name = "incremental_sync", version = "1.0.0")]
async fn incremental_sync(ctx: WorkflowContext, input: SyncInput) -> Result<SyncResult> {
    let watermark = ctx.execute_task(get_watermark, input.source_name.clone()).await?;
    let since = watermark.unwrap_or_else(|| "2024-01-01T00:00:00Z".into());

    let records = ctx.execute_task(fetch_new_records, (
        input.source_name.clone(), since.clone(),
    )).await?;

    if records.is_empty() {
        return Ok(SyncResult { records_synced: 0, new_watermark: since });
    }

    let transformed = ctx.execute_task(transform_records, records.clone()).await?;
    ctx.execute_task(load_records, (input.target_table, transformed)).await?;

    let new_watermark = records.last().unwrap().created_at.clone();
    ctx.execute_task(save_watermark, (input.source_name, new_watermark.clone())).await?;

    Ok(SyncResult { records_synced: records.len(), new_watermark })
}

@task(name="transform_records")
async def transform_records(ctx: TaskContext, records: list[dict]) -> list[dict]:
    return [{
        "id": r["id"],
        "amount": r["amount"] / 100,  # cents to dollars
        "currency": r["currency"].upper(),
        "status": map_status(r["status"]),
        "customer_id": r["customer_id"],
        "created_at": r["created_at"],
    } for r in records]

async def load_records(ctx: TaskContext, input: dict) -> int:
    return await db.upsert_rows(input["table"], input["records"])

@task(name="save_watermark")
async def save_watermark(ctx: TaskContext, input: dict) -> None:
    await db.query("""
        INSERT INTO sync_state (source, watermark, updated_at)
        VALUES ($1, $2, NOW())
        ON CONFLICT (source) DO UPDATE SET watermark = $2, updated_at = NOW()
    """, [input["source"], input["watermark"]])

@workflow(name="incremental_sync", version="1.0.0")
async def incremental_sync(ctx: WorkflowContext, input: SyncInput) -> SyncResult:
    watermark = await ctx.execute_task(get_watermark, source=input.source_name)
    since = watermark or "2024-01-01T00:00:00Z"

    records = await ctx.execute_task(fetch_new_records, input={
        "source": input.source_name, "since": since,
    })

    if not records:
        return SyncResult(records_synced=0, new_watermark=since)

    transformed = await ctx.execute_task(transform_records, records=records)
    await ctx.execute_task(load_records, input={
        "table": input.target_table, "records": transformed,
    })

    new_watermark = records[-1]["created_at"]
    await ctx.execute_task(save_watermark, input={
        "source": input.source_name, "watermark": new_watermark,
    })

    return SyncResult(records_synced=len(records), new_watermark=new_watermark)
