## Monitoring and Observability

### Understanding Workflow State

Orcher provides rich visibility into workflow execution through the workflow journal and metadata tables. Every workflow moves through a small set of states — and monitoring is mostly about watching for the two that mean trouble.

**Query Workflow Status:**

```sql
-- Get workflow overview
SELECT
    id,
    workflow_name,
    status,
    created_at,
    completed_at,
    attempt,
    error
FROM workflows
WHERE status IN ('running', 'failed')
ORDER BY created_at DESC
LIMIT 100;
```

**Monitor Task Performance:**

```sql
-- Find slow tasks across all workflows
SELECT
    task_name,
    COUNT(*) as execution_count,
    AVG(EXTRACT(EPOCH FROM (completed_at - created_at))) as avg_duration_seconds,
    MAX(EXTRACT(EPOCH FROM (completed_at - created_at))) as max_duration_seconds,
    PERCENTILE_CONT(0.95) WITHIN GROUP (ORDER BY EXTRACT(EPOCH FROM (completed_at - created_at))) as p95_duration_seconds
FROM workflow_execution_journal
WHERE status = 'completed'
    AND completed_at > NOW() - INTERVAL '24 hours'
GROUP BY task_name
ORDER BY avg_duration_seconds DESC;
```

**Track Failure Rates:**

```sql
-- Calculate task failure rates
SELECT
    task_name,
    COUNT(*) FILTER (WHERE status = 'completed') as success_count,
    COUNT(*) FILTER (WHERE status = 'failed') as failure_count,
    ROUND(
        100.0 * COUNT(*) FILTER (WHERE status = 'failed') / COUNT(*),
        2
    ) as failure_rate_percent
FROM workflow_execution_journal
WHERE created_at > NOW() - INTERVAL '7 days'
GROUP BY task_name
HAVING COUNT(*) > 10
ORDER BY failure_rate_percent DESC;
```

```rust
use tracing::{info, warn, error, instrument};

#[workflow(name = "process_payment")]
#[instrument(skip(ctx), fields(workflow_id = %ctx.workflow_id()))]
pub async fn process_payment(
    ctx: WorkflowContext,
    payment: PaymentInput,
) -> Result<PaymentResult> {
    info!(
        user_id = %payment.user_id,
        amount = payment.amount,
        "Starting payment processing"
    );

    // Log decision points
    let fraud_score = ctx.execute_task(check_fraud, payment.user_id.clone()).await?;

    if fraud_score > 0.8 {
        warn!(
            user_id = %payment.user_id,
            fraud_score = fraud_score,
            "Payment flagged as fraudulent"
        );
        return Err(anyhow::anyhow!("Payment flagged"));
    }

    info!(fraud_score = fraud_score, "Fraud check passed");

    // Execute payment
    let charge = ctx.execute_task(charge_stripe, payment.clone()).await?;

    info!(
        charge_id = %charge.id,
        amount = payment.amount,
        "Payment charged successfully"
    );

    Ok(PaymentResult {
        charge_id: charge.id,
        status: "completed".to_string(),
        receipt_url: charge.receipt_url,
    })
}

#[task]
#[instrument]
async fn charge_stripe(ctx: TaskContext, payment: PaymentInput) -> Result<StripeCharge> {
    info!("Calling Stripe API");

    let start = std::time::Instant::now();
    let charge = make_stripe_call(payment).await?;
    let duration = start.elapsed();

    info!(
        duration_ms = duration.as_millis(),
        charge_id = %charge.id,
        "Stripe API call completed"
    );

    Ok(charge)
}
```

```typescript
import { logger } from './logger';

@Tasks()
class PaymentTasksLogged {
  @Task()
  async checkFraud(ctx: TaskContext, userId: string): Promise<number> { /* ... */ }

  @Task()
  async chargeStripe(ctx: TaskContext, payment: PaymentInput): Promise<StripeCharge> {
    logger.info('Calling Stripe API', { paymentMethod: payment.paymentMethod });

    const startTime = Date.now();
    const charge = await makeStripeCall(payment);
    const duration = Date.now() - startTime;

    logger.info('Stripe API call completed', {
      chargeId: charge.id,
      durationMs: duration,
    });

    return charge;
  }
}

const paymentTasksLogged = createTaskRefs(PaymentTasksLogged);

@Workflow({ name: 'process_payment', version: '1.0.0' })
class ProcessPaymentLogged {
  async run(ctx: WorkflowContext, payment: PaymentInput): Promise<PaymentResult> {
    const workflowId = ctx.workflowId();

    logger.info('Starting payment processing', {
      workflowId,
      userId: payment.userId,
      amount: payment.amount,
    });

    const fraudScore = await ctx.executeTask(paymentTasksLogged.checkFraud, payment.userId);

    if (fraudScore > 0.8) {

        workflowId,
        userId: payment.userId,
        fraudScore,
      });
      throw new Error('Payment flagged');
    }

    logger.info('Fraud check passed', { workflowId, fraudScore });

    const charge = await ctx.executeTask(paymentTasksLogged.chargeStripe, payment);

    logger.info('Payment charged successfully', {
      workflowId,
      chargeId: charge.id,
      amount: payment.amount,
    });

    return {
      chargeId: charge.id,
      status: 'completed',
      receiptUrl: charge.receiptUrl,
    };
  }
}
```

**Logging Best Practices:**

✓ Always include `workflow_id` for correlation
✓ Log at decision points (if/else branches)
✓ Log external API calls with duration
✓ Log business-significant events
✗ Don't log sensitive data (credit cards, passwords)
✗ Don't log inside tight loops

### Metrics and Alerting

**Key Metrics to Track:**

1. **Workflow Completion Rate**
   ```sql
   SELECT
       workflow_name,
       COUNT(*) FILTER (WHERE status = 'completed') * 100.0 / COUNT(*) as completion_rate
   FROM workflows
   WHERE created_at > NOW() - INTERVAL '1 hour'
   GROUP BY workflow_name;
   ```

2. **Workflow Duration (P50, P95, P99)**
   ```sql
   SELECT
       workflow_name,
       PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY EXTRACT(EPOCH FROM (completed_at - created_at))) as p50_seconds,
       PERCENTILE_CONT(0.95) WITHIN GROUP (ORDER BY EXTRACT(EPOCH FROM (completed_at - created_at))) as p95_seconds,
       PERCENTILE_CONT(0.99) WITHIN GROUP (ORDER BY EXTRACT(EPOCH FROM (completed_at - created_at))) as p99_seconds
   FROM workflows
   WHERE status = 'completed'
       AND created_at > NOW() - INTERVAL '1 hour'
   GROUP BY workflow_name;
   ```

3. **Queue Depth (Pending Workflows)**
   ```sql
   SELECT COUNT(*) as pending_workflows
   FROM workflows
   WHERE status = 'pending';
   ```

4. **Retry Rate**
   ```sql
   SELECT
       workflow_name,
       AVG(attempt) as avg_attempts,
       MAX(attempt) as max_attempts
   FROM workflows
   WHERE created_at > NOW() - INTERVAL '1 hour'
   GROUP BY workflow_name;
   ```

**Alerting Rules:**

- **Critical**: Workflow failure rate > 5% over 5 minutes
- **Warning**: P95 duration > 2x baseline
- **Warning**: Queue depth > 1000 workflows
- **Critical**: Any workflow stuck for > 1 hour

```rust
#[cfg(test)]
mod integration_tests {
    use super::*;
    use orcher::test_utils::TestOrcher;

    #[tokio::test]
    async fn test_payment_workflow_success() {
        // Setup test Orcher instance with in-memory DB
        let orcher = TestOrcher::new().await;

        // Register workflow
        orcher.register_workflow(process_payment).await;

        // Start workflow
        let payment = PaymentInput {
            user_id: "user_123".to_string(),
            amount: 5000,
            currency: "usd".to_string(),
            payment_method: "pm_test_card".to_string(),
        };

        let workflow_id = orcher
            .start_workflow("process_payment", payment)
            .await
            .unwrap();

        // Wait for completion (with timeout)
        let result = orcher
            .wait_for_workflow::<PaymentResult>(workflow_id, Duration::from_secs(30))
            .await
            .unwrap();

        // Assert result
        assert_eq!(result.status, "completed");
        assert!(!result.charge_id.is_empty());

        // Verify journal entries
        let journal = orcher.get_workflow_journal(workflow_id).await.unwrap();
        assert_eq!(journal.len(), 4); // validate, fraud_check, charge, save
        assert!(journal.iter().all(|e| e.status == "completed"));
    }

    #[tokio::test]
    async fn test_payment_workflow_fraud_rejection() {
        let orcher = TestOrcher::new().await;
        orcher.register_workflow(process_payment).await;

        // Use high-risk user ID that triggers fraud detection
        let payment = PaymentInput {
            user_id: "high_risk_user".to_string(),
            amount: 5000,
            currency: "usd".to_string(),
            payment_method: "pm_test_card".to_string(),
        };

        let workflow_id = orcher
            .start_workflow("process_payment", payment)
            .await
            .unwrap();

        // Wait for failure
        let result = orcher
            .wait_for_workflow::<PaymentResult>(workflow_id, Duration::from_secs(30))
            .await;

        assert!(result.is_err());

        // Verify workflow failed at fraud check
        let workflow = orcher.get_workflow(workflow_id).await.unwrap();
        assert_eq!(workflow.status, "failed");
        assert!(workflow.error.unwrap().contains("fraud"));
    }

    #[tokio::test]
    async fn test_workflow_retry_on_transient_failure() {
        let orcher = TestOrcher::new().await;
        orcher.register_workflow(process_payment).await;

        // Inject transient failure in stripe task
        orcher.inject_task_failure("charge_stripe", 2); // Fail first 2 attempts

        let payment = PaymentInput {
            user_id: "user_123".to_string(),
            amount: 5000,
            currency: "usd".to_string(),
            payment_method: "pm_test_card".to_string(),
        };

        let workflow_id = orcher
            .start_workflow("process_payment", payment)
            .await
            .unwrap();

        // Should eventually succeed after retries
        let result = orcher
            .wait_for_workflow::<PaymentResult>(workflow_id, Duration::from_secs(60))
            .await
            .unwrap();

        assert_eq!(result.status, "completed");

        // Verify retry attempts
        let workflow = orcher.get_workflow(workflow_id).await.unwrap();
        assert!(workflow.attempt >= 3);
    }
}
```

## Performance Optimization

### Database Tuning

**Index Optimization:**

```sql
-- Essential indexes for Orcher tables
CREATE INDEX CONCURRENTLY idx_workflows_status_created
ON workflows(status, created_at)
WHERE status IN ('pending', 'running');

CREATE INDEX CONCURRENTLY idx_workflow_execution_journal_lookup
ON workflow_execution_journal(workflow_id, step_id);

CREATE INDEX CONCURRENTLY idx_workflow_execution_journal_task_perf
ON workflow_execution_journal(task_name, status, created_at, completed_at)
WHERE completed_at IS NOT NULL;

-- Index for queue polling (critical for performance)
CREATE INDEX CONCURRENTLY idx_workflows_queue_poll
ON workflows(status, created_at)
WHERE status = 'pending';
```

**Connection Pooling:**

The database connection pool belongs to the **orchestrator**, not to your workflow or worker code — there is no `Orcher.builder().with_pool()` API. Workers are stateless gRPC clients and never open a database connection; only the orchestrator talks to PostgreSQL. You size the pool where the orchestrator runs, via environment variables:

```bash
# Orchestrator connection pool (set per orchestrator node)
ORCHER_PG_MAX_CONNECTIONS=80   # max pooled connections (default 80)
ORCHER_PG_MIN_CONNECTIONS=20   # warm connections kept open (default 20)
```

**Sizing rule** — every orchestrator node holds up to `ORCHER_PG_MAX_CONNECTIONS`, so when you run *N* nodes against one database:

```
N × ORCHER_PG_MAX_CONNECTIONS  ≤  PostgreSQL max_connections   (default 100)
```

To run several orchestrator nodes against a single PostgreSQL, lower `ORCHER_PG_MAX_CONNECTIONS` per node (e.g. three nodes at 30 each = 90 < 100). For higher aggregate concurrency without exhausting PostgreSQL, put **PgBouncer** in front of the database in transaction-pooling mode — the orchestrator's connections then multiplex over a small number of real backend connections. If you need more headroom directly, raise `max_connections` in `postgresql.conf`.

Because the pool is an orchestrator concern, adding **workers** never adds database connections — workers scale execution capacity over gRPC, decoupled from the database. See the [scaling guide](/guides/learning/platform-engineers/scaling) for the full worker-vs-orchestrator model.

**Worker Configuration:**

A worker is its own process: a stateless gRPC client of the orchestrator. It opens **no** database connection — it pulls work over gRPC and runs your registered workflows and tasks. Scale out by running more replicas (as in the Deployment above); scale *up* within a process with the `max_concurrent_*` knobs.

```rust
use orcher::prelude::*;
use std::time::Duration;

#[tokio::main]
async fn main() -> Result<()> {
    // Workflows and tasks are auto-registered from their macros — no DB pool here.
    WorkerBuilder::new()
        .server_url(
            std::env::var("ORCHER_SERVER_URL")
                .unwrap_or_else(|_| "http://orchestrator:50051".to_string()),
        )
        .namespace("production")
        .task_queue("default")
        .max_concurrent_workflows(20) // workflows running at once in this process
        .max_concurrent_tasks(40)     // tasks running at once in this process
        .heartbeat_interval(Duration::from_secs(30)) // liveness to the orchestrator
        .build()
        .await?
        .run() // runs until shutdown
        .await?;

    Ok(())
}
```

### Health Checks

**Liveness and Readiness Probes:**

```rust
use axum::{Router, routing::get, Json};
use serde_json::json;

async fn health_check() -> Json<serde_json::Value> {
    Json(json!({
        "status": "healthy",
        "timestamp": chrono::Utc::now().to_rfc3339(),
    }))
}

async fn readiness_check(
    Extension(pool): Extension<PgPool>,
) -> Json<serde_json::Value> {
    // Check database connectivity
    let db_ok = sqlx::query("SELECT 1").fetch_one(&pool).await.is_ok();

    // Check queue depth
    let queue_depth: i64 = sqlx::query_scalar(
        "SELECT COUNT(*) FROM workflows WHERE status = 'pending'"
    )
    .fetch_one(&pool)
    .await
    .unwrap_or(0);

    let ready = db_ok && queue_depth < 10000;

    Json(json!({
        "ready": ready,
        "database": db_ok,
        "queue_depth": queue_depth,
        "timestamp": chrono::Utc::now().to_rfc3339(),
    }))
}

// Add to router
let app = Router::new()
    .route("/health", get(health_check))
    .route("/ready", get(readiness_check));
```

### Graceful Shutdown

Handle shutdown signals properly to avoid interrupting workflows:

```rust
use tokio::signal;

async fn shutdown_signal() {
    let ctrl_c = async {
        signal::ctrl_c()
            .await
            .expect("failed to install Ctrl+C handler");
    };

    #[cfg(unix)]
    let terminate = async {
        signal::unix::signal(signal::unix::SignalKind::terminate())
            .expect("failed to install signal handler")
            .recv()
            .await;
    };

    #[cfg(not(unix))]
    let terminate = std::future::pending::<()>();

    tokio::select! {
        _ = ctrl_c => {},
        _ = terminate => {},
    }

    info!("Shutdown signal received, starting graceful shutdown...");
}

#[tokio::main]
async fn main() -> Result<()> {
    let orcher = Orcher::new().await?;

    // Run server with graceful shutdown
    let server = axum::Server::bind(&"0.0.0.0:8080".parse()?)
        .serve(app.into_make_service())
        .with_graceful_shutdown(shutdown_signal());

    // Shutdown workers when signal received
    tokio::select! {
        result = server => {
            result?;
        }
        _ = shutdown_signal() => {
            info!("Stopping workers...");
            orcher.stop_workers().await?;
            info!("Workers stopped gracefully");
        }
    }

    Ok(())
}
```
